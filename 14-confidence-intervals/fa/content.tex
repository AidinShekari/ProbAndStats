\begin{PSbox}[unbreakable]{obj}{اهداف یادگیری}

پس از تکمیل این ماژول، دانشجویان قادر خواهند بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  بازه‌های اطمینان را به‌درستی تفسیر کنند (تفسیر فراوانی‌گرا)
\item
  بازه‌های اطمینان برای میانگین بسازند (واریانس معلوم و نامعلوم)
\item
  توزیع \lr{\mbox{t}} را برای نمونه‌های کوچک به کار ببرند
\item
  بازه‌های اطمینان برای واریانس بسازند
\item
  اندازه نمونه لازم را تعیین کنند
\item
  از تصورهای نادرست رایج درباره بازه‌های اطمینان پرهیز کنند
\end{enumerate}

\end{PSbox}

\PSsection{14.1}{مقدمه: چرا برآورد نقطه‌ای کافی نیست}

یک برآورد نقطه‌ای (مثلاً \lr{\mbox{\ensuremath{\bar{X}} = 3.47}}) هیچ اطلاعاتی درباره \textbf{دقت} نمی‌دهد. باید عدم‌قطعیت را کمّی کنیم:

«میانگین واقعی \PSnum{3.47} برآورد می‌شود، \textbf{با تقریبی} به اندازه‌ای.»

یک \textbf{بازه اطمینان} بازه‌ای فراهم می‌کند که با سطح اطمینان معین، پارامتر واقعی را در بر می‌گیرد.

\PSsection{14.2}{تفسیر درست بازه‌های اطمینان}

\PSsubsection{\lr{\mbox{\PSyes{}}} تفسیر درست (فراوانی‌گرا)}

\begin{PSquote}

«اگر این آزمایش را بارها تکرار کنیم و هر بار یک بازه اطمینان 95\% محاسبه کنیم، آنگاه تقریباً 95\% از آن بازه‌ها شامل پارامتر واقعی \lr{\mbox{\ensuremath{\mu}}} خواهند بود.»

\end{PSquote}

این \textbf{رویه} نرخ موفقیت 95\% دارد. هر بازه مشخص یا شامل \lr{\mbox{\ensuremath{\mu}}} است یا نیست.

\PSsubsection{\lr{\mbox{\PSno{}}} تفسیر نادرست}

\begin{PSquote}

{«احتمال 95\% وجود دارد که \lr{\mbox{\ensuremath{\mu}}} داخل این بازه خاص \lr{\mbox{[2.1, 4.8]}} باشد.»}

\end{PSquote}

\textbf{چرا نادرست است:} \lr{\mbox{\ensuremath{\mu}}} یک عدد ثابت (غیرتصادفی) است. یا در \lr{\mbox{[2.1, 4.8]}} است یا نیست \lr{—} هیچ احتمالی درباره آن وجود ندارد. \textbf{بازه} تصادفی است (از نمونه‌ای به نمونه دیگر تغییر می‌کند)، نه پارامتر.

\PSsubsection{قیاس}

بازه‌های اطمینان را مانند یک تور ماهیگیری در نظر بگیرید:

\begin{itemize}
\tightlist
\item
  پیش از پرتاب: شانس 95\% که تور شما ماهی را بگیرد (این رویه 95\% مواقع کار می‌کند)
\item
  پس از پرتاب: ماهی یا گرفته شده یا نشده \lr{—} دیگر احتمالی وجود ندارد
\item
  ماهی \lr{\mbox{(\ensuremath{\mu})}} حرکت نمی‌کند؛ تور شما (بازه اطمینان) حرکت می‌کند
\end{itemize}

\PSsection{14.3}{بازه اطمینان برای میانگین \lr{—} واریانس معلوم (بازه \lr{\mbox{Z}})}

\PSsubsection{صورت‌بندی}

داده: \lr{\mbox{X\textsubscript{1}}}، \lr{…}، \lr{\mbox{X\textsubscript{n}}} مستقل و هم‌توزیع از \lr{\mbox{N(\ensuremath{\mu}, \ensuremath{\sigma}\textsuperscript{2})}} با \lr{\mbox{\ensuremath{\sigma}\textsuperscript{2}}} \textbf{معلوم}.

\PSsubsection{استخراج}

\lr{\ensuremath{\bar{X}} \textasciitilde{} N(\ensuremath{\mu}, \ensuremath{\sigma}\textsuperscript{2}/n)}، بنابراین \lr{Z = (\ensuremath{\bar{X}} - \ensuremath{\mu})/(\ensuremath{\sigma}/\ensuremath{\surd}n) \textasciitilde{} N(0,1)}

\lr{P(-z\textsubscript{\ensuremath{\alpha}/2} \ensuremath{\le} Z \ensuremath{\le} z\textsubscript{\ensuremath{\alpha}/2}) = 1 - \ensuremath{\alpha}}

با حل برای \lr{\mbox{\ensuremath{\mu}}}:\PSdisplay{P\left(\bar{X} - z_{\alpha/2}\frac{\sigma}{\sqrt{n}} \leq \mu \leq \bar{X} + z_{\alpha/2}\frac{\sigma}{\sqrt{n}}\right) = 1 - \alpha}

\PSsubsection{فرمول}

\PSdisplay{\PSmtext{CI: } \bar{X} \pm z_{\alpha/2} \cdot \frac{\sigma}{\sqrt{n}}}

\PSsubsection{مقادیر رایج \lr{\mbox{z}}}

{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[c]{@{}ccc@{}}
\toprule\noalign{}
\rowcolor{PSnavy}\PSth{سطح اطمینان} & \PSth{\lr{\mbox{\ensuremath{\alpha}}}} & \PSth{\lr{\mbox{z\textsubscript{\ensuremath{\alpha}/2}}}} \\
\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
90\% & \PSnum{0.10} & \PSnum{1.645} \\
95\% & \PSnum{0.05} & \PSnum{1.960} \\
99\% & \PSnum{0.01} & \PSnum{2.576} \\
\end{longtable}
}

\begin{PSbox}[unbreakable]{ex}{مثال}

اندازه‌گیری‌های نویز: \lr{\mbox{n = 36}}، \lr{\mbox{\ensuremath{\bar{X}} = 0.52V}}، \lr{\mbox{\ensuremath{\sigma} = 0.12V}} (معلوم از مشخصات تجهیز).

بازه اطمینان 95\%: \lr{0.52 \ensuremath{\pm} 1.96 \ensuremath{\times} 0.12/\ensuremath{\surd}36 = 0.52 \ensuremath{\pm} 0.0392 = \textbf{\lr{\mbox{[0.481, 0.559] V}}}}

\end{PSbox}

\PSsection{14.4}{بازه اطمینان برای میانگین \lr{—} واریانس نامعلوم (بازه \lr{\mbox{t}})}

\begin{PSbox}[unbreakable]{prob}{مسیله}

معمولاً \lr{\mbox{\ensuremath{\sigma}}} نامعلوم است. آن را با \lr{\mbox{S}} برآورد می‌کنیم. اما \lr{\mbox{\ensuremath{\bar{X}}/(S/\ensuremath{\surd}n)}} از \lr{\mbox{N(0,1)}} پیروی نمی‌کند.

\end{PSbox}

\PSsubsection{توزیع \lr{\mbox{t}} استیودنت}

\PSdisplay{T = \frac{\bar{X} - \mu}{S/\sqrt{n}} \sim t(n-1)}

توزیع \lr{\mbox{t}}:

\begin{itemize}
\tightlist
\item
  متقارن حول 0
\item
  دنباله‌های سنگین‌تر از \lr{\mbox{N(0,1)}} (عدم‌قطعیت بیشتر چون \lr{\mbox{S}} برآورد شده است)
\item
  درجات آزادی: \lr{\mbox{\ensuremath{\nu} = n - 1}}
\item
  وقتی \lr{\mbox{\ensuremath{\nu} \ensuremath{\to} \ensuremath{\infty}}}: \lr{\mbox{t(\ensuremath{\nu}) \ensuremath{\to} N(0,1)}}
\item
  برای \lr{\mbox{\ensuremath{\nu} \ensuremath{\ge} 30}}: تقریباً \lr{\mbox{t \ensuremath{\approx} z}}
\end{itemize}

\PSsubsection{فرمول}

\PSdisplay{\PSmtext{CI: } \bar{X} \pm t_{\alpha/2, n-1} \cdot \frac{S}{\sqrt{n}}}

\begin{PSbox}[unbreakable]{ex}{مثال}

تأخیر قطعه: \lr{\mbox{n = 15}}، \lr{\mbox{\ensuremath{\bar{X}} = 2.34 ns}}، \lr{\mbox{S = 0.45 ns}}.

بازه اطمینان 95\%: \lr{\mbox{t\textsubscript{0.025, 14} = 2.145}}

\lr{\mbox{CI}}: \lr{2.34 \ensuremath{\pm} 2.145 \ensuremath{\times} 0.45/\ensuremath{\surd}15 = 2.34 \ensuremath{\pm} 0.249 = \textbf{\lr{\mbox{[2.09, 2.59] ns}}}}

(پهن‌تر از بازه \lr{\mbox{Z}} به‌دلیل عدم‌قطعیت اضافی در \lr{\mbox{S}}.)

\end{PSbox}

\PSsection{14.5}{بازه اطمینان برای واریانس}

\PSsubsection{صورت‌بندی}

داده از \lr{\mbox{N(\ensuremath{\mu}, \ensuremath{\sigma}\textsuperscript{2})}}. بازه اطمینان برای \lr{\mbox{\ensuremath{\sigma}\textsuperscript{2}}} مطلوب است.

\PSsubsection{توزیع کلیدی}

\PSdisplay{(n-1)S^2/\sigma^2 \sim \chi^2(n-1)}

\PSsubsection{فرمول}

\PSdisplay{\left[\frac{(n-1)S^2}{\chi^2_{\alpha/2, n-1}}, \quad \frac{(n-1)S^2}{\chi^2_{1-\alpha/2, n-1}}\right]}

\begin{PSbox}[unbreakable]{ex}{مثال}

توان نویز: \lr{\mbox{n = 20}}، \lr{\mbox{S\textsuperscript{2} = 0.045 V\textsuperscript{2}}}.

بازه اطمینان 95\% برای \lr{\mbox{\ensuremath{\sigma}\textsuperscript{2}}}: \lr{\ensuremath{\chi}\textsuperscript{2}\textsubscript{0.025, 19} = 32.852}، \lr{\ensuremath{\chi}\textsuperscript{2}\textsubscript{0.975, 19} = 8.907}

\lr{\mbox{CI}}: [\lr{\mbox{(19)(0.045)/32.852}}، \lr{\mbox{(19)(0.045)/8.907}}] = \lr{\mbox{[0.026, 0.096] V\textsuperscript{2}}}

بازه اطمینان 95\% برای \lr{\mbox{\ensuremath{\sigma}}}: \lr{[\ensuremath{\surd}0.026, \ensuremath{\surd}0.096] = [0.161, 0.310] V}

\end{PSbox}

\PSsection{14.6}{بازه اطمینان برای نسبت‌ها}

\PSsubsection{صورت‌بندی}

\lr{\mbox{n}} آزمایش، \lr{\mbox{\ensuremath{\hat{x}} = k/n}} موفقیت. برآورد \lr{\mbox{p}} = نسبت واقعی.

\PSsubsection{فرمول نمونه بزرگ (بازه والد)}

\PSdisplay{\hat{p} \pm z_{\alpha/2}\sqrt{\frac{\hat{p}(1-\hat{p})}{n}}}

\begin{PSbox}[unbreakable]{ex}{مثال: نرخ خطای بیت}

\PSnum{100,000} بیت ارسال شد، 53 خطا مشاهده شد. \lr{\ensuremath{\hat{p}} = 53/100000 = 5.3\ensuremath{\times}10\textsuperscript{-4}}.

بازه اطمینان 95\%: \lr{5.3\ensuremath{\times}10\textsuperscript{-4} \ensuremath{\pm} 1.96\ensuremath{\surd}(5.3\ensuremath{\times}10\textsuperscript{-4} \ensuremath{\times} 0.9995 / 100000) = 5.3\ensuremath{\times}10\textsuperscript{-4} \ensuremath{\pm} 1.43\ensuremath{\times}10\textsuperscript{-4}}

\lr{\mbox{CI}}: \lr{\textbf{\lr{[3.9\ensuremath{\times}10\textsuperscript{-4}, 6.7\ensuremath{\times}10\textsuperscript{-4}]}}}

\end{PSbox}

\PSsection{14.8}{تعیین اندازه نمونه}

\PSsubsection{پرسش: چند اندازه‌گیری لازم دارم؟}

برای پهنای بازه اطمینان \lr{\mbox{\ensuremath{\le} 2E}} (حاشیه خطای \lr{\mbox{E}}):

\PSdisplay{n \geq \left(\frac{z_{\alpha/2} \cdot \sigma}{E}\right)^2}

\begin{PSbox}[unbreakable]{ex}{مثال}

می‌خواهیم میانگین تأخیر را با اطمینان 95\% و در محدوده \lr{\mbox{\ensuremath{\pm}0.5ms}} برآورد کنیم. انتظار \lr{\mbox{\ensuremath{\sigma} \ensuremath{\approx} 3ms}}.

\lr{n \ensuremath{\ge} (1.96 \ensuremath{\times} 3 / 0.5)\textsuperscript{2} = (11.76)\textsuperscript{2} = 138.3 \ensuremath{\to}} به \textbf{\lr{\mbox{n = 139}} اندازه‌گیری} نیاز است

\end{PSbox}
