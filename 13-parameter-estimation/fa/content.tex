\begin{PSbox}[unbreakable]{obj}{اهداف یادگیری}

پس از تکمیل این ماژول، دانشجویان قادر خواهند بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  میان برآوردگر (قاعده) و برآورد (عدد) تمایز قایل شوند
\item
  ویژگی‌های برآوردگر را ارزیابی کنند: اریبی، سازگاری، کارایی، \lr{\mbox{MSE}}
\item
  روش گشتاورها را به کار ببرند
\item
  برآورد بیشینه درست‌نمایی \lr{\mbox{(MLE)}} را به کار ببرند
\item
  برآوردگرها را با استفاده از \lr{\mbox{MSE}} و توازن اریبی-واریانس مقایسه کنند
\end{enumerate}

\end{PSbox}

\PSsection{13.1}{مقدمه}

داده \lr{\mbox{X\textsubscript{1}}}، \lr{…}، \lr{\mbox{X\textsubscript{n}}} را از توزیعی با پارامتر نامعلوم \lr{\mbox{\ensuremath{\theta}}} مشاهده می‌کنیم. \textbf{برآورد پارامتر} روشی برای استنباط \lr{\mbox{\ensuremath{\theta}}} از داده به ما می‌دهد.

مثال‌های مهندسی:

\begin{itemize}
\tightlist
\item
  برآورد واریانس نویز \lr{\mbox{\ensuremath{\sigma}\textsuperscript{2}}} از نمونه‌های ثبت‌شده
\item
  برآورد نرخ خرابی \lr{\mbox{\ensuremath{\lambda}}} از عمر قطعات
\item
  برآورد بهره کانال از قدرت سیگنال دریافتی
\item
  برآورد سطح میانگین سیگنال از اندازه‌گیری‌ها
\end{itemize}

\PSsubsection{نکته کلیدی}

\textbf{برآوردگر} یک متغیر تصادفی است \lr{—} توزیع نمونه‌گیری دارد.\PSbr{}\textbf{برآورد} یک تحقق از آن متغیر تصادفی است.

\textbf{قیاس:} «میانگین بگیر» برآوردگر است (قاعده). «میانگین برابر \PSnum{3.47} بود» برآورد است (نتیجه).

\PSsection{13.3}{ویژگی‌های برآوردگرها}

\PSsubsection{اریبی}

\PSdisplay{B(\hat{\theta}) = E[\hat{\theta}] - \theta}

\begin{itemize}
\tightlist
\item
  \textbf{نااریب:} \lr{\mbox{B(\ensuremath{\hat{\theta}}) = 0}}، یعنی \lr{\mbox{E[\ensuremath{\hat{\theta}}] = \ensuremath{\theta}}} (به‌طور میانگین درست)
\item
  \textbf{اریب:} \lr{E[\ensuremath{\hat{\theta}}] \ensuremath{\neq} \ensuremath{\theta}} (خطای نظام‌مند)
\end{itemize}

مثال‌ها:

\begin{itemize}
\tightlist
\item
  \lr{\mbox{\ensuremath{\bar{X}}}} برای \lr{\mbox{\ensuremath{\mu}}} نااریب است: \lr{\mbox{E[\ensuremath{\bar{X}}] = \ensuremath{\mu} \PSyes{}}}
\item
  \lr{\mbox{S\textsuperscript{2}}} (با \lr{\mbox{n-1}}) برای \lr{\mbox{\ensuremath{\sigma}\textsuperscript{2}}} نااریب است: \lr{E[S\textsuperscript{2}] = \ensuremath{\sigma}\textsuperscript{2} \PSyes{}}
\item
  \lr{\mbox{S}} (انحراف معیار نمونه) برای \lr{\mbox{\ensuremath{\sigma}}} اریب است: \lr{\mbox{E[S] \ensuremath{\neq} \ensuremath{\sigma}}} (اریبی اندک به سمت پایین)
\end{itemize}

\PSsubsection{سازگاری}

\lr{\mbox{\ensuremath{\hat{\theta}}\textsubscript{n}}} \textbf{سازگار} است اگر \lr{\ensuremath{\hat{\theta}}\textsubscript{n} \ensuremath{\to} \ensuremath{\theta}} در احتمال وقتی \lr{\mbox{n \ensuremath{\to} \ensuremath{\infty}}}.

معنای عملی: با داده کافی، برآوردگر به‌طور دلخواه به مقدار واقعی نزدیک می‌شود.

شرط کافی: اگر اریبی \lr{\mbox{\ensuremath{\to} 0}} و واریانس \lr{\mbox{\ensuremath{\to} 0}} وقتی \lr{\mbox{n \ensuremath{\to} \ensuremath{\infty}}}، آنگاه سازگار است.

\PSsubsection{کارایی}

در میان تمام برآوردگرهای نااریب، برآوردگر \textbf{کارا} کمترین واریانس را دارد.

\textbf{کران پایین کرامر-رایو} \lr{\mbox{(CRLB)}} کمترین واریانس ممکن را می‌دهد:\PSdisplay{\PSmtext{Var}(\hat{\theta}) \geq \frac{1}{nI(\theta)}}

که در آن \lr{\mbox{I(\ensuremath{\theta})}} = اطلاعات فیشر = \lr{-E[\ensuremath{\partial}\textsuperscript{2}ln f(X;\ensuremath{\theta})/\ensuremath{\partial}\ensuremath{\theta}\textsuperscript{2}]}.

برآوردگری که به \lr{\mbox{CRLB}} برسد \textbf{کارا} یا \textbf{کم‌واریانس نااریب \lr{\mbox{(MVU)}}} نامیده می‌شود.

\PSsubsection{میانگین مربعات خطا \lr{\mbox{(MSE)}}}

\PSdisplay{\PSmtext{MSE}(\hat{\theta}) = E[(\hat{\theta} - \theta)^2] = \PSmtext{Var}(\hat{\theta}) + [B(\hat{\theta})]^2}

\textbf{\lr{\mbox{MSE}} = واریانس + اریبی\lr{\mbox{\textsuperscript{2}}}}

این همان \textbf{توازن اریبی-واریانس} است: گاهی یک برآوردگر کمی اریب با واریانس بسیار کمتر، \lr{\mbox{MSE}} کوچک‌تری از یک برآوردگر نااریب دارد.

\PSsection{13.4}{روش گشتاورها \lr{\mbox{(MoM)}}}

\PSsubsection{اصل}

گشتاورهای جامعه را برابر گشتاورهای نمونه قرار دهید، سپس برای پارامترها حل کنید.

\begin{itemize}
\tightlist
\item
  گشتاور مرتبه \lr{\mbox{k}} جامعه: \lr{\ensuremath{\mu}’\textsubscript{k} = E[X\textsuperscript{k}]}
\item
  گشتاور مرتبه \lr{\mbox{k}} نمونه: \lr{m’\textsubscript{k} = (1/n)\ensuremath{\Sigma}X\textsubscript{i}\textsuperscript{k}}
\end{itemize}

\PSsubsection{رویه}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  پارامترها را بر حسب گشتاورهای جامعه بیان کنید
\item
  گشتاورهای جامعه را با گشتاورهای نمونه جایگزین کنید
\item
  برای برآوردهای پارامتر حل کنید
\end{enumerate}

\begin{PSbox}[unbreakable]{ex}{مثال: توزیع نمایی}

داده از \lr{\mbox{Exp(\ensuremath{\lambda})}}. گشتاور جامعه: \lr{\mbox{E[X] = 1/\ensuremath{\lambda}}}.\PSbr{}برابر قرار دهید: \lr{(1/n)\ensuremath{\Sigma}X\textsubscript{i} = 1/\ensuremath{\hat{\lambda}} \ensuremath{\to} \ensuremath{\hat{\lambda}}\textsubscript{MoM} = 1/\ensuremath{\bar{X}}}

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال: توزیع گاوسی}

داده از \lr{\mbox{N(\ensuremath{\mu}, \ensuremath{\sigma}\textsuperscript{2})}}:

\begin{itemize}
\tightlist
\item
  گشتاور اول: \lr{\mbox{\ensuremath{\hat{\mu}} = \ensuremath{\bar{X}}}}
\item
  گشتاور مرکزی دوم: \lr{\ensuremath{\hat{\sigma}}\textsuperscript{2} = (1/n)\ensuremath{\Sigma}(X\textsubscript{i} - \ensuremath{\bar{X}})\textsuperscript{2}} (توجه: اریب، از \lr{\mbox{n}} استفاده می‌کند نه \lr{\mbox{n-1}})
\end{itemize}

\end{PSbox}

\PSsubsection{مزایا/معایب}

\lr{\mbox{\PSyes{}}} ساده، همیشه قابل اعمال، جواب‌های فرم بسته اغلب در دسترس\PSbr{}\lr{\mbox{\PSno{}}} همیشه کارا نیست، می‌تواند برآوردهای اریب تولید کند، ممکن است مقادیر نامعتبر پارامتر بدهد

\PSsection{13.5}{برآورد بیشینه درست‌نمایی \lr{\mbox{(MLE)}}}

\PSsubsection{اصل}

مقدار پارامتری را انتخاب کنید که داده مشاهده‌شده را \textbf{محتمل‌ترین} می‌کند.

\PSsubsection{تابع درست‌نمایی}

با داده \lr{\mbox{x\textsubscript{1}}}، \lr{…}، \lr{\mbox{x\textsubscript{n}}} از \lr{\mbox{f(x; \ensuremath{\theta})}}:

\PSdisplay{L(\theta) = \prod_{i=1}^n f(x_i; \theta)}

\PSsubsection{لگاریتم درست‌نمایی (معمولاً آسان‌تر)}

\PSdisplay{\ell(\theta) = \ln L(\theta) = \sum_{i=1}^n \ln f(x_i; \theta)}

\PSsubsection{رویه \lr{\mbox{MLE}}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  لگاریتم درست‌نمایی \lr{\mbox{\ensuremath{\ell}(\ensuremath{\theta})}} را بنویسید
\item
  مشتق بگیرید: \lr{\mbox{d\ensuremath{\ell}/d\ensuremath{\theta} = 0}}
\item
  برای \lr{\mbox{\ensuremath{\hat{\theta}}\textsubscript{MLE}}} حل کنید
\item
  بررسی کنید که بیشینه است \lr{(d\textsuperscript{2}\ensuremath{\ell}/d\ensuremath{\theta}\textsuperscript{2} \textless{} 0)}
\end{enumerate}

\PSsubsection{ویژگی‌های \lr{\mbox{MLE}}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{سازگار} (به مقدار واقعی همگرا می‌شود)
\item
  \textbf{به‌طور مجانبی نااریب} (اریبی \lr{\mbox{\ensuremath{\to} 0}} وقتی \lr{\mbox{n \ensuremath{\to} \ensuremath{\infty}}})
\item
  \textbf{به‌طور مجانبی کارا} (برای \lr{\mbox{n}} بزرگ به \lr{\mbox{CRLB}} می‌رسد)
\item
  \textbf{ناوردا:} اگر \lr{\mbox{\ensuremath{\hat{\theta}}}} برآوردگر \lr{\mbox{MLE}} برای \lr{\mbox{\ensuremath{\theta}}} باشد، آنگاه \lr{\mbox{g(\ensuremath{\hat{\theta}})}} برآوردگر \lr{\mbox{MLE}} برای \lr{\mbox{g(\ensuremath{\theta})}} است
\end{enumerate}

\begin{PSbox}[unbreakable]{ex}{مثال: \lr{\mbox{MLE}} برای میانگین گاوسی}

داده \lr{\mbox{X\textsubscript{1}}}، \lr{…}، \lr{X\textsubscript{n} \textasciitilde{} N(\ensuremath{\mu}, \ensuremath{\sigma}\textsuperscript{2})} با \lr{\mbox{\ensuremath{\sigma}\textsuperscript{2}}} معلوم.

\lr{\ensuremath{\ell}(\ensuremath{\mu}) = -(n/2)ln(2\ensuremath{\pi}\ensuremath{\sigma}\textsuperscript{2}) - (1/(2\ensuremath{\sigma}\textsuperscript{2}))\ensuremath{\Sigma}(x\textsubscript{i} - \ensuremath{\mu})\textsuperscript{2}}

\lr{d\ensuremath{\ell}/d\ensuremath{\mu} = (1/\ensuremath{\sigma}\textsuperscript{2})\ensuremath{\Sigma}(x\textsubscript{i} - \ensuremath{\mu}) = 0 \ensuremath{\to} \ensuremath{\hat{\mu}}\textsubscript{MLE} = \ensuremath{\bar{X}}}

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال: \lr{\mbox{MLE}} برای نرخ نمایی}

داده \lr{\mbox{X\textsubscript{1}}}، \lr{…}، \lr{X\textsubscript{n} \textasciitilde{} Exp(\ensuremath{\lambda})}.

\lr{\ensuremath{\ell}(\ensuremath{\lambda}) = n\ensuremath{\cdot}ln(\ensuremath{\lambda}) - \ensuremath{\lambda}\ensuremath{\cdot}\ensuremath{\Sigma}x\textsubscript{i}}

\lr{d\ensuremath{\ell}/d\ensuremath{\lambda} = n/\ensuremath{\lambda} - \ensuremath{\Sigma}x\textsubscript{i} = 0 \ensuremath{\to} \ensuremath{\hat{\lambda}}\textsubscript{MLE} = n/\ensuremath{\Sigma}x\textsubscript{i} = 1/\ensuremath{\bar{X}}}

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال: \lr{\mbox{MLE}} برای واریانس گاوسی}

داده \lr{\mbox{X\textsubscript{1}}}، \lr{…}، \lr{X\textsubscript{n} \textasciitilde{} N(\ensuremath{\mu}, \ensuremath{\sigma}\textsuperscript{2})}، \lr{\mbox{\ensuremath{\mu}}} معلوم.

\lr{\ensuremath{\ell}(\ensuremath{\sigma}\textsuperscript{2}) = -(n/2)ln(2\ensuremath{\pi}\ensuremath{\sigma}\textsuperscript{2}) - (1/(2\ensuremath{\sigma}\textsuperscript{2}))\ensuremath{\Sigma}(x\textsubscript{i} - \ensuremath{\mu})\textsuperscript{2}}

\lr{d\ensuremath{\ell}/d(\ensuremath{\sigma}\textsuperscript{2}) = -n/(2\ensuremath{\sigma}\textsuperscript{2}) + (1/(2\ensuremath{\sigma}\textsuperscript{4}))\ensuremath{\Sigma}(x\textsubscript{i} - \ensuremath{\mu})\textsuperscript{2} = 0}

\lr{\ensuremath{\hat{\sigma}}\textsuperscript{2}\textsubscript{MLE} = (1/n)\ensuremath{\Sigma}(x\textsubscript{i} - \ensuremath{\mu})\textsuperscript{2}}

توجه: با \lr{\mbox{\ensuremath{\mu}}} نامعلوم، \lr{\ensuremath{\hat{\sigma}}\textsuperscript{2}\textsubscript{MLE} = (1/n)\ensuremath{\Sigma}(x\textsubscript{i} - \ensuremath{\bar{x}})\textsuperscript{2}} \lr{—} اریب! (از \lr{\mbox{n}} استفاده می‌کند، نه \lr{\mbox{n-1}})

\end{PSbox}

\PSsection{13.6}{مثال‌های مهندسی}

\PSsubsection{برآورد واریانس نویز}

\textbf{مسیله:} \lr{\mbox{n}} نمونه نویز اندازه‌گیری کنید. توان نویز \lr{\mbox{\ensuremath{\sigma}\textsuperscript{2}}} را برآورد کنید.

\textbf{داده:} \lr{\mbox{x\textsubscript{1}}}، \lr{…}، \lr{\mbox{x\textsubscript{n}}} (نمونه‌های ولتاژ نویز، با فرض میانگین صفر)

\lr{\textbf{\lr{\mbox{MLE:}}} \ensuremath{\hat{\sigma}}\textsuperscript{2} = (1/n)\ensuremath{\Sigma}x\textsubscript{i}\textsuperscript{2}} (اریب با ضریب \lr{\mbox{(n-1)/n}})\PSbr{}\textbf{نااریب:} \lr{S\textsuperscript{2} = (1/(n-1))\ensuremath{\Sigma}(x\textsubscript{i} - \ensuremath{\bar{x}})\textsuperscript{2}} یا اگر \lr{\mbox{\ensuremath{\mu}=0}} معلوم باشد: \lr{(1/n)\ensuremath{\Sigma}x\textsubscript{i}\textsuperscript{2}} در واقع برای \lr{E[X\textsuperscript{2}]=\ensuremath{\sigma}\textsuperscript{2}} نااریب است

\PSsubsection{برآورد نرخ خرابی}

\textbf{مسیله:} 20 قطعه تا خرابی آزمایش می‌شوند. عمرها: \lr{\mbox{t\textsubscript{1}}}، \lr{…}، \lr{\mbox{t\textsubscript{20}}}.

\textbf{مدل:} \lr{\mbox{T \textasciitilde{} Exp(\ensuremath{\lambda})}}، پارامتر \lr{\mbox{\ensuremath{\lambda}}} (نرخ خرابی).

\lr{\textbf{\lr{\mbox{MLE:}}} \ensuremath{\hat{\lambda}} = 20/\ensuremath{\Sigma}t\textsubscript{i} = 1/\ensuremath{\bar{T}}}

اگر \lr{\mbox{\ensuremath{\bar{T}} = 500}} ساعت باشد: \lr{\mbox{\ensuremath{\hat{\lambda}} = 1/500 = 0.002}} خرابی در ساعت.

\PSsubsection{برآورد پارامتر کانال}

\textbf{مسیله:} کانال محوشدگی رایلی. نمونه‌های توان مشاهده‌شده: \lr{\mbox{p\textsubscript{1}}}، \lr{…}، \lr{\mbox{p\textsubscript{n}}}.

\textbf{مدل:} توان \lr{\mbox{P \textasciitilde{} Exp(1/\ensuremath{\Omega})}}، که در آن \lr{\mbox{\ensuremath{\Omega} = E[P]}} = توان میانگین.

\lr{\textbf{\lr{\mbox{MLE:}}} \ensuremath{\hat{\Omega}} = \ensuremath{\bar{P}} = (1/n)\ensuremath{\Sigma}p\textsubscript{i}}

\PSsection{13.7}{مقایسه برآوردگرها}

\PSsubsection{توازن اریبی-واریانس}

دو برآوردگر برای \lr{\mbox{\ensuremath{\sigma}\textsuperscript{2}}}:

\begin{itemize}
\tightlist
\item
  \lr{\ensuremath{\hat{\theta}}\textsubscript{1} = S\textsuperscript{2}} (نااریب، واریانس = \lr{\mbox{2\ensuremath{\sigma}\textsuperscript{4}/(n-1)}})
\item
  \lr{\ensuremath{\hat{\theta}}\textsubscript{2} = (1/n)\ensuremath{\Sigma}(X\textsubscript{i}-\ensuremath{\bar{X}})\textsuperscript{2}} (اریب با \lr{\mbox{-\ensuremath{\sigma}\textsuperscript{2}/n}}، واریانس = \lr{2\ensuremath{\sigma}\textsuperscript{4}(n-1)/n\textsuperscript{2}})
\end{itemize}

\lr{MSE(\ensuremath{\hat{\theta}}\textsubscript{1}) = 2\ensuremath{\sigma}\textsuperscript{4}/(n-1)}\PSbr{}\lr{MSE(\ensuremath{\hat{\theta}}\textsubscript{2}) = 2\ensuremath{\sigma}\textsuperscript{4}(n-1)/n\textsuperscript{2} + \ensuremath{\sigma}\textsuperscript{4}/n\textsuperscript{2} = \ensuremath{\sigma}\textsuperscript{4}(2n-1)/n\textsuperscript{2}}

برای هر \lr{\mbox{n \ensuremath{\ge} 2}}: \lr{MSE(\ensuremath{\hat{\theta}}\textsubscript{2}) \textless{} MSE(\ensuremath{\hat{\theta}}\textsubscript{1})}! برآوردگر اریب \lr{\mbox{MLE}} در واقع خطای کل کوچک‌تری دارد.

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

داده از \lr{\mbox{N(\ensuremath{\mu}, 9)}}: \lr{\mbox{\ensuremath{\bar{x}} = 4.2}}، \lr{\mbox{n = 25}}.

\begin{enumerate}
\def\labelenumi{(\PSalph{enumi})}
\tightlist
\item
  \lr{\mbox{MLE}} مربوط به \lr{\mbox{\ensuremath{\mu}}} را بنویسید. (ب) \lr{\mbox{Var(\ensuremath{\hat{\mu}})}} چقدر است؟ (ج) آیا \lr{\mbox{\ensuremath{\bar{X}}}} برای \lr{\mbox{\ensuremath{\mu}}} کارا است؟
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\PSalph{enumi})}
\tightlist
\item
  \lr{\ensuremath{\hat{\mu}}\textsubscript{MLE} = \ensuremath{\bar{X}} = 4.2}
\item
  \lr{Var(\ensuremath{\bar{X}}) = \ensuremath{\sigma}\textsuperscript{2}/n = 9/25 = 0.36}
\item
  \lr{\mbox{CRLB}} برای \lr{\mbox{N(\ensuremath{\mu},\ensuremath{\sigma}\textsuperscript{2})}}: \lr{1/(n\ensuremath{\cdot}I(\ensuremath{\mu})) = \ensuremath{\sigma}\textsuperscript{2}/n = 0.36}. \lr{\mbox{\ensuremath{\bar{X}}}} به \lr{\mbox{CRLB}} می‌رسد \lr{\mbox{\ensuremath{\to}}} بله، کارا است.
\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2}

  \lr{\ensuremath{\bar{X}} = (120+350+200+480+90+560+310+175+420+280)/10 = 298.5.}\PSbr{}\lr{\mbox{\ensuremath{\hat{\lambda}} = 1/298.5 = 0.00335}} خرابی در ساعت.
