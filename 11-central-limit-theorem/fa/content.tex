\PSsection{11.1}{مقدمه: چرا گاوسی همه‌جا هست؟}

توزیع گاوسی در موارد زیر ظاهر می‌شود:

\begin{itemize}
\tightlist
\item
  نویز حرارتی، خطاهای اندازه‌گیری، رواداری‌های ساخت، تداخل تجمعی، نوسان‌های سهام
\end{itemize}

\textbf{دلیل:} همه این کمیت‌ها جمع‌ها (یا میانگین‌ها)ی بسیاری از اثرهای تصادفی کوچک و مستقل هستند. \lr{CLT} همگرایی به گاوسی را صرف‌نظر از توزیع زیربنایی تضمین می‌کند.

\PSsection{11.2}{قانون اعداد بزرگ \lr{(LLN)}}

\PSsubsection{قانون ضعیف اعداد بزرگ}

برای \lr{X\textsubscript{1}, X\textsubscript{2}, …, X\textsubscript{n}} مستقل و هم‌توزیع \lr{(i.i.d.)} با میانگین \lr{\ensuremath{\mu}} و واریانس متناهی \lr{\ensuremath{\sigma}\textsuperscript{2}}:

\PSdisplay{\bar{X}_n = \frac{1}{n}\sum_{i=1}^n X_i \xrightarrow{P} \mu \quad \PSmtext{as } n \to \infty}

(«همگرایی در احتمال»: \lr{P(\textbar{}\ensuremath{\bar{X}}\textsubscript{n} - \ensuremath{\mu}\textbar{} \textgreater{} \ensuremath{\varepsilon}) \ensuremath{\to} 0} برای هر \lr{\ensuremath{\varepsilon} \textgreater{} 0})

\PSsubsection{قانون قوی اعداد بزرگ}

\PSdisplay{\bar{X}_n \xrightarrow{a.s.} \mu \quad \PSmtext{as } n \to \infty}

(«همگرایی تقریباً مطمین»: \lr{P(\ensuremath{\bar{X}}\textsubscript{n} \ensuremath{\to} \ensuremath{\mu}) = 1})

\PSsubsection{اهمیت مهندسی}

\begin{itemize}
\tightlist
\item
  \textbf{میانگین نمونه با رشد اندازه نمونه به میانگین واقعی همگرا می‌شود}
\item
  برآورد میانگین‌ها از داده را توجیه می‌کند
\item
  بنیاد شبیه‌سازی مونت‌کارلو (شبیه‌سازی بارها \lr{\ensuremath{\to}} میانگین همگرا می‌شود)
\item
  این گزاره را توجیه می‌کند: «اگر به‌قدر کافی اندازه‌گیری کنم، میانگین من به مقدار واقعی نزدیک خواهد بود»
\end{itemize}

\PSsubsection{با چه سرعتی؟}

بر اساس چبیشف: \lr{P(\textbar{}\ensuremath{\bar{X}}\textsubscript{n} - \ensuremath{\mu}\textbar{} \textgreater{} \ensuremath{\varepsilon}) \ensuremath{\le} \ensuremath{\sigma}\textsuperscript{2}/(n\ensuremath{\varepsilon}\textsuperscript{2})}

برای اطمینان 95\% از این‌که خطا \lr{\textless{}} \lr{\ensuremath{\varepsilon}} باشد: به \lr{n \ensuremath{\ge} \ensuremath{\sigma}\textsuperscript{2}/(0.05 \ensuremath{\cdot} \ensuremath{\varepsilon}\textsuperscript{2})} نیاز است

\PSsection{11.3}{قضیه حد مرکزی \lr{(CLT)}}

\PSsubsection{بیان صوری}

برای \lr{X\textsubscript{1}, X\textsubscript{2}, …, X\textsubscript{n}} مستقل و هم‌توزیع \lr{(i.i.d.)} با میانگین \lr{\ensuremath{\mu}} و واریانس متناهی \lr{\ensuremath{\sigma}\textsuperscript{2}}:

\PSdisplay{Z_n = \frac{\bar{X}_n - \mu}{\sigma/\sqrt{n}} = \frac{\sum_{i=1}^n X_i - n\mu}{\sigma\sqrt{n}} \xrightarrow{d} N(0,1)}

\PSsubsection{شکل عملی}

برای \lr{n} بزرگ:\PSdisplay{\bar{X}_n \approx N\left(\mu, \frac{\sigma^2}{n}\right)}

\PSdisplay{S_n = \sum_{i=1}^n X_i \approx N(n\mu, n\sigma^2)}

\PSsubsection{شرایط}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{X\textsubscript{1}, …, X\textsubscript{n}} مستقل هستند
\item
  هم‌توزیع هستند (با \lr{CLT} لیپانوف/لینده‌برگ می‌توان این شرط را سست کرد)
\item
  میانگین متناهی \lr{\ensuremath{\mu}} و واریانس متناهی \lr{\ensuremath{\sigma}\textsuperscript{2}}
\item
  \lr{n} «به‌قدر کافی بزرگ» است (قاعده سرانگشتی: \lr{n \ensuremath{\ge} 30}، اما به شکل توزیع بستگی دارد)
\end{enumerate}

\PSsubsection{آنچه \lr{CLT} نیاز ندارد}

\begin{itemize}
\tightlist
\item
  نیاز ندارد \lr{X\textsubscript{i}} گاوسی باشد
\item
  نیاز ندارد \lr{X\textsubscript{i}} پیوسته باشد
\item
  نیاز ندارد هیچ توزیع خاصی داشته باشد
\end{itemize}

\PSsection{11.5}{نمایش بصری}

\PSsubsection{جمع متغیرهای تصادفی یکنواخت \lr{\ensuremath{\to}} گاوسی}

\begin{itemize}
\tightlist
\item
  \lr{n=1}: یکنواخت (مسطح)
\item
  \lr{n=2}: مثلثی
\item
  \lr{n=3}: شروع به زنگوله‌ای‌شدن می‌کند
\item
  \lr{n=5}: تقریباً گاوسی
\item
  \lr{n=12}: عملاً از گاوسی قابل تفکیک نیست
\end{itemize}

\PSsubsection{جمع متغیرهای تصادفی نمایی \lr{\ensuremath{\to}} گاوسی}

\begin{itemize}
\tightlist
\item
  \lr{n=1}: نمایی (به‌شدت چوله به راست)
\item
  \lr{n=3}: همچنان چوله اما کمتر \lr{(Gamma(3))}
\item
  \lr{n=10}: تقریباً متقارن، زنگوله‌ای
\item
  \lr{n=30}: اساساً گاوسی
\end{itemize}

\PSsubsection{دوجمله‌ای \lr{\ensuremath{\to}} تقریب نرمال}

\lr{Binomial(n, p) \ensuremath{\approx} N(np, np(1-p))} برای \lr{n} بزرگ

این همان \lr{CLT} اعمال‌شده به جمع آزمایش‌های برنولی است!

\begin{itemize}
\tightlist
\item
  تصحیح پیوستگی: \lr{P(X \ensuremath{\le} k) \ensuremath{\approx} \ensuremath{\Phi}((k + 0.5 - np)/(\ensuremath{\surd}(np(1-p))))}
\end{itemize}

\PSsection{11.6}{اهمیت مهندسی}

\PSsubsection{چرا نویز حرارتی گاوسی است}

نویز حرارتی = اثر تجمعی میلیاردها الکترون که به‌طور تصادفی حرکت می‌کنند. سهم هر الکترون ناچیز و مستقل است \lr{\ensuremath{\to} CLT} اعمال می‌شود \lr{\ensuremath{\to}} نویز گاوسی است.

\PSsubsection{چرا تداخل تجمعی گاوسی است}

در یک شبکه سلولی، تداخل کل = جمع سیگنال‌های بسیاری از کاربران مستقل. \lr{CLT \ensuremath{\to}} تداخل \lr{\ensuremath{\approx}} گاوسی (برای کاربران زیاد).

\PSsubsection{چرا خطاهای اندازه‌گیری گاوسی هستند}

خطای کل = جمع بسیاری از منابع خطای کوچک و مستقل (کالیبراسیون، کوانتیزاسیون، گرمایی، لرزش و \lr{…}) \lr{\ensuremath{\to} CLT \ensuremath{\to}} خطا \lr{\ensuremath{\approx}} گاوسی.

\PSsubsection{چرا تغییرات ساخت گاوسی هستند}

مقدار مقاومت = مقدار نامی + جمع بسیاری از تغییرات کوچک فرایند (دوپینگ، حکاکی، دما و \lr{…}) \lr{\ensuremath{\to} CLT \ensuremath{\to}} مقدار \lr{\ensuremath{\approx}} گاوسی حول مقدار نامی.

\PSsection{11.8}{تقریب‌زنی با استفاده از \lr{CLT}}

\PSsubsection{رویه کلی}

برای تقریب \lr{P(S\textsubscript{n} \ensuremath{\le} x)} که در آن \lr{S\textsubscript{n} = X\textsubscript{1} + … + X\textsubscript{n}}:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{\ensuremath{\mu} = E[X\textsubscript{i}]}، \lr{\ensuremath{\sigma}\textsuperscript{2} = Var(X\textsubscript{i})} را محاسبه کنید
\item
  \lr{S\textsubscript{n} \ensuremath{\approx} N(n\ensuremath{\mu}, n\ensuremath{\sigma}\textsuperscript{2})}
\item
  \lr{P(S\textsubscript{n} \ensuremath{\le} x) \ensuremath{\approx} \ensuremath{\Phi}((x - n\ensuremath{\mu})/(\ensuremath{\sigma}\ensuremath{\surd}n))}
\end{enumerate}

\PSsubsection{تصحیح پیوستگی (برای متغیرهای تصادفی گسسته)}

هنگام تقریب‌زدن یک جمع گسسته با گاوسی:

\begin{itemize}
\tightlist
\item
  \lr{P(X \ensuremath{\le} k) \ensuremath{\approx} \ensuremath{\Phi}((k + 0.5 - n\ensuremath{\mu})/(\ensuremath{\sigma}\ensuremath{\surd}n))}
\item
  \lr{P(X = k) \ensuremath{\approx} \ensuremath{\Phi}((k + 0.5 - n\ensuremath{\mu})/(\ensuremath{\sigma}\ensuremath{\surd}n)) - \ensuremath{\Phi}((k - 0.5 - n\ensuremath{\mu})/(\ensuremath{\sigma}\ensuremath{\surd}n))}
\end{itemize}

\begin{PSbox}[unbreakable]{ex}{مثال: خطاهای بسته}

1000 بسته، احتمال خطای هر یک \lr{0.02. X} = مجموع خطاها.\PSbr{}دقیق: \lr{X \textasciitilde{} Binomial(1000, 0.02)}\PSbr{}تقریب \lr{CLT: X \ensuremath{\approx} N(20, 19.6)}

\lr{P(X \textgreater{} 30) \ensuremath{\approx} 1 - \ensuremath{\Phi}((30.5 - 20)/\ensuremath{\surd}19.6) = 1 - \ensuremath{\Phi}(2.37) = 0.0089}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

یک کارخانه پیچ تولید می‌کند. طول \lr{X} \lr{\textasciitilde{}} توزیعی با \lr{\ensuremath{\mu} = 10cm} و \lr{\ensuremath{\sigma} = 0.2cm}. یک دسته 100 پیچی اندازه‌گیری می‌شود.

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  توزیع تقریبی \lr{\ensuremath{\bar{X}}} چیست؟ \lr{(b) P(\ensuremath{\bar{X}} \textgreater{} 10.05)}؟ \lr{(c) P(9.96 \textless{} \ensuremath{\bar{X}} \textless{} 10.04)}؟
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  \lr{\ensuremath{\bar{X}} \textasciitilde{} N(10, 0.04/100) = N(10, 0.0004)}، \lr{\ensuremath{\sigma}\textsubscript{X}̄ = 0.02}
\item
  \lr{P(\ensuremath{\bar{X}} \textgreater{} 10.05) = P(Z \textgreater{} 2.5) = 0.0062}
\item
  \lr{P(9.96 \textless{} \ensuremath{\bar{X}} \textless{} 10.04) = \ensuremath{\Phi}(2) - \ensuremath{\Phi}(-2) = 0.9545}
\end{enumerate}

\end{PSbox}
