\PSsection{5.2}{تبدیل‌های خطی: \lr{Y = aX + b}}

\PSsubsection{نتیجه}

اگر \lr{X} دارای \lr{PDF f\textsubscript{X}(x)} باشد، آنگاه \lr{Y = aX + b} دارای:

\[f_Y(y) = \frac{1}{|a|} f_X\left(\frac{y-b}{a}\right)\]

\PSsubsection{میانگین و واریانس}

\begin{itemize}
\tightlist
\item
  \lr{E[Y] = aE[X] + b}
\item
  \lr{Var(Y) = a\textsuperscript{2}Var(X)}
\end{itemize}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: تقویت‌کننده}

سیگنال ورودی \lr{X \textasciitilde{} N(0, \ensuremath{\sigma}\textsuperscript{2})} از تقویت‌کننده‌ای با بهره \lr{G = 10} و آفست \lr{DC} یعنی \lr{V\textsubscript{0} = 1.5V} می‌گذرد.

خروجی: \lr{Y = 10X + 1.5}

\begin{itemize}
\tightlist
\item
  \lr{Y \textasciitilde{} N(10\ensuremath{\cdot}0 + 1.5, 10\textsuperscript{2}\ensuremath{\cdot}\ensuremath{\sigma}\textsuperscript{2}) = N(1.5, 100\ensuremath{\sigma}\textsuperscript{2})}
\item
  توان نویز خروجی به اندازه \lr{G\textsuperscript{2} = 100} افزایش یافته است
\end{itemize}

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: حسگر دما}

خروجی حسگر \lr{V = 0.01\ensuremath{\cdot}T + 0.5 (V} بر حسب ولت، \lr{T} بر حسب \lr{\ensuremath{^{\circ}}C})

اگر \lr{T \textasciitilde{} N(25, 4): V \textasciitilde{} N(0.01\ensuremath{\times}25 + 0.5, 0.01\textsuperscript{2}\ensuremath{\times}4) = N(0.75, 0.0004)}

\end{PSbox}

\PSsection{5.3}{تبدیل‌های یکنوای عمومی}

\PSsubsection{روش \lr{CDF} (کلی‌ترین روش)}

برای \lr{Y = g(X)}، \lr{f\textsubscript{Y}(y)} را به این ترتیب بیابید:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  بنویسید \lr{F\textsubscript{Y}(y) = P(Y \ensuremath{\le} y) = P(g(X) \ensuremath{\le} y)}
\item
  نامساوی را برای \lr{X} حل کنید
\item
  بر حسب \lr{F\textsubscript{X}} بیان کنید
\item
  مشتق بگیرید تا \lr{f\textsubscript{Y}(y)} به دست آید
\end{enumerate}

\PSsubsection{فرمول برای \lr{g} یکنوا}

اگر \lr{g} اکیداً صعودی با وارون \lr{g\textsuperscript{-1}} باشد:\PSbr{}\[f_Y(y) = f_X(g^{-1}(y)) \cdot \frac{d}{dy}[g^{-1}(y)]\]

اگر \lr{g} اکیداً نزولی باشد:\PSbr{}\[f_Y(y) = f_X(g^{-1}(y)) \cdot \left|\frac{d}{dy}[g^{-1}(y)]\right|\]

ترکیبی (برای هر دو کار می‌کند):\PSbr{}\[f_Y(y) = \frac{f_X(x)}{|g'(x)|}\bigg|_{x=g^{-1}(y)}\]

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: \lr{SNR} بر حسب \lr{dB}}

\lr{SNR} خطی: \lr{X \textasciitilde{} Exponential(1)} (توان محوشدگی رایلی نرمال‌شده).\PSbr{}\lr{SNR} بر حسب \lr{dB: Y = 10\ensuremath{\cdot}log\textsubscript{10}(X)}

\begin{itemize}
\tightlist
\item
  \lr{g(x) = 10\ensuremath{\cdot}log\textsubscript{10}(x)}، \lr{g’(x) = 10/(x\ensuremath{\cdot}ln10)}
\item
  \lr{g\textsuperscript{-1}(y) = 10\textsuperscript{y/10}}
\end{itemize}

\[f_Y(y) = \frac{f_X(10^{y/10})}{10/(10^{y/10} \cdot \ln 10)} = \frac{\ln 10}{10} \cdot 10^{y/10} \cdot e^{-10^{y/10}}\]

این توزیع گاوسی نیست \lr{—} دنباله چپ بلندتری دارد (فروافت‌های عمیق).

\end{PSbox}

\PSsection{5.4}{تبدیل‌های غیرخطی: حالت غیریکنوا}

\PSsubsection{هنگامی که \lr{Y = g(X)} یک‌به‌یک نیست}

اگر چند مقدار \lr{x} به همان \lr{y} نگاشت شوند، روی تمام جواب‌ها جمع کنید:

\[f_Y(y) = \sum_{i} \frac{f_X(x_i)}{|g'(x_i)|}\]

که در آن \lr{x\textsubscript{1}, x\textsubscript{2}, …} تمام ریشه‌های \lr{g(x) = y} هستند.

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: توان از ولتاژ \lr{(Y = X\textsuperscript{2})}}

ولتاژ \lr{X \textasciitilde{} N(0, \ensuremath{\sigma}\textsuperscript{2})}. توان: \lr{Y = X\textsuperscript{2}}.

برای \lr{y \textgreater{} 0}، جواب‌ها \lr{x = +\ensuremath{\surd}y} و \lr{x = -\ensuremath{\surd}y} هستند. \lr{g’(x) = 2x}.

\[f_Y(y) = \frac{f_X(\sqrt{y})}{2\sqrt{y}} + \frac{f_X(-\sqrt{y})}{2\sqrt{y}} = \frac{1}{\sqrt{2\pi}\sigma} \cdot \frac{e^{-y/(2\sigma^2)}}{\sqrt{y}}\]

این یک \textbf{توزیع کای‌دو با 1 درجه آزادی} است (مقیاس‌شده با \lr{\ensuremath{\sigma}\textsuperscript{2}}).

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: یکسوساز تمام‌موج \lr{(Y = \textbar{}X\textbar{})}}

ورودی \lr{X \textasciitilde{} N(0, \ensuremath{\sigma}\textsuperscript{2})}. خروجی \lr{Y = \textbar{}X\textbar{}}.

برای \lr{y \textgreater{} 0: x = +y} و \lr{x = -y} جواب هستند. \lr{\textbar{}g’(x)\textbar{} = 1}.

\[f_Y(y) = f_X(y) + f_X(-y) = \frac{2}{\sigma\sqrt{2\pi}} e^{-y^2/(2\sigma^2)}, \quad y \geq 0\]

این توزیع \textbf{نرمال تاشده} (نیم‌نرمال) است.

\end{PSbox}

\PSsection{5.5}{روش ژاکوبی}

\PSsubsection{رویه برای \lr{Y = g(X)}}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  تبدیل \lr{y = g(x)} را مشخص کنید
\item
  وارون را بیابید: \lr{x = g\textsuperscript{-1}(y)}
\item
  ژاکوبی را محاسبه کنید: \lr{J = \textbar{}dx/dy\textbar{} = \textbar{}d[g\textsuperscript{-1}(y)]/dy\textbar{}}
\item
  اعمال کنید: \lr{f\textsubscript{Y}(y) = f\textsubscript{X}(g\textsuperscript{-1}(y)) \ensuremath{\cdot} \textbar{}J\textbar{}}
\end{enumerate}

\begin{PSbox}[unbreakable]{ex}{مثال گام‌به‌گام: نمایی گاوسی}

\lr{X \textasciitilde{} N(\ensuremath{\mu}, \ensuremath{\sigma}\textsuperscript{2}). Y = e\textsuperscript{x}} (تبدیل لگ‌نرمال).

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{g(x) = e\textsuperscript{x}} (یکنوای صعودی)
\item
  \lr{x = ln(y)}، معتبر برای \lr{y \textgreater{} 0}
\item
  \lr{J = \textbar{}dx/dy\textbar{} = 1/y}
\item
  \lr{f\textsubscript{Y}(y) = f\textsubscript{X}(ln y) \ensuremath{\cdot} (1/y) = (1/(y\ensuremath{\sigma}\ensuremath{\surd}(2\ensuremath{\pi}))) \ensuremath{\cdot} exp(-(ln y - \ensuremath{\mu})\textsuperscript{2}/(2\ensuremath{\sigma}\textsuperscript{2}))}
\end{enumerate}

این همان \textbf{توزیع لگ‌نرمال} است \lr{—} بسیاری از کمیت‌های مهندسی را مدل می‌کند (سایه‌اندازی در بی‌سیم، قیمت سهام، عمر قطعات همراه با فرسودگی).

\end{PSbox}

\PSsection{5.6}{کاربرد مهندسی: رایلی از گاوسی}

\PSsubsection{استخراج توزیع رایلی}

در مخابرات، سیگنال دریافتی مؤلفه‌های هم‌فاز \lr{(I)} و مربعی \lr{(Q)} دارد:

\begin{itemize}
\tightlist
\item
  \lr{I \textasciitilde{} N(0, \ensuremath{\sigma}\textsuperscript{2})}، \lr{Q \textasciitilde{} N(0, \ensuremath{\sigma}\textsuperscript{2})}، مستقل
\end{itemize}

پوش: \lr{R = \ensuremath{\surd}(I\textsuperscript{2} + Q\textsuperscript{2})}

با استفاده از تبدیل \lr{(I, Q) \ensuremath{\to} (R, \ensuremath{\theta})} در مختصات قطبی:

\begin{itemize}
\tightlist
\item
  \lr{I = R\ensuremath{\cdot}cos(\ensuremath{\theta})}، \lr{Q = R\ensuremath{\cdot}sin(\ensuremath{\theta})}
\item
  ژاکوبی: \lr{\textbar{}\ensuremath{\partial}(I,Q)/\ensuremath{\partial}(R,\ensuremath{\theta})\textbar{} = R}
\end{itemize}

\lr{PDF} مشترک \lr{(R, \ensuremath{\theta})}:\PSbr{}\lr{f\textsubscript{R,\ensuremath{\Theta}}(r,\ensuremath{\theta}) = f\textsubscript{I,Q}(r\ensuremath{\cdot}cos\ensuremath{\theta}, r\ensuremath{\cdot}sin\ensuremath{\theta}) \ensuremath{\cdot} r = (r/(2\ensuremath{\pi}\ensuremath{\sigma}\textsuperscript{2})) \ensuremath{\cdot} e\textsuperscript{-r\textsuperscript{2}/(2\ensuremath{\sigma}\textsuperscript{2})}}

حاشیه‌سازی روی \lr{\ensuremath{\theta} \ensuremath{\in} [0, 2\ensuremath{\pi})}:

\[f_R(r) = \frac{r}{\sigma^2} e^{-r^2/(2\sigma^2)}, \quad r \geq 0\]

این همان \textbf{توزیع رایلی} است \lr{—} بنیادی برای مدل‌سازی کانال محوشدگی بی‌سیم.

\PSsection{5.8}{مسایل تمرینی}

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

سیگنال \lr{X \textasciitilde{} Uniform[0, 1]}. خروجی \lr{Y = -2\ensuremath{\cdot}ln(X). PDF} مربوط به \lr{Y} را بیابید و توزیع را تشخیص دهید.

\PSsolution{} روش \lr{CDF: F\textsubscript{Y}(y) = P(-2ln(X) \ensuremath{\le} y) = P(X \ensuremath{\ge} e\textsuperscript{-y/2}) = 1 - e\textsuperscript{-y/2}} برای \lr{y \ensuremath{\ge} 0}.\PSbr{}\lr{f\textsubscript{Y}(y) = (1/2)e\textsuperscript{-y/2} \ensuremath{\to} Y \textasciitilde{} Exponential(\ensuremath{\beta} = 2)}. (این همان روش \lr{CDF} وارون برای تولید توزیع نمایی است!)

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2}

ولتاژ نویز \lr{X \textasciitilde{} N(0, \ensuremath{\sigma}\textsuperscript{2})} با \lr{\ensuremath{\sigma} = 2V}. توان تلف‌شده در \lr{R = 50\ensuremath{\Omega}: P = X\textsuperscript{2}/R}.

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  \lr{E[P]} را بیابید. \lr{(b) PDF} مربوط به \lr{P} را بیابید. \lr{(c) P(P \textgreater{} 0.2W)} را بیابید.
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  \lr{E[P] = E[X\textsuperscript{2}]/R = \ensuremath{\sigma}\textsuperscript{2}/R = 4/50 = 0.08W}
\item
  \lr{P = X\textsuperscript{2}/50}. فرض کنید \lr{W = X\textsuperscript{2} \textasciitilde{} \ensuremath{\sigma}\textsuperscript{2}\ensuremath{\cdot}\ensuremath{\chi}\textsuperscript{2}(1)}. آنگاه \lr{P = W/50. f\textsubscript{P}(p) = 50\ensuremath{\cdot}f\textsubscript{W}(50p) = (50/(2\ensuremath{\sigma}\textsuperscript{2}))\ensuremath{\cdot}(50p/\ensuremath{\sigma}\textsuperscript{2})\textsuperscript{\lr{(-1/2)\ensuremath{\cdot}e}}(-50p/(2\ensuremath{\sigma}\textsuperscript{2}))} برای \lr{p \textgreater{} 0}.
\item
  از \lr{MATLAB} استفاده کنید: \lr{\PScode{1 - chi2cdf(0.2*50/4, 1)} = 1 - chi2cdf(2.5, 1) \ensuremath{\approx} 0.114}
\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 3}

\lr{X \textasciitilde{} Exponential(\ensuremath{\lambda} = 1). Y = \ensuremath{\surd}X}. مقدار \lr{f\textsubscript{Y}(y)} را بیابید.

\PSsolution{} \lr{g(x) = \ensuremath{\surd}x \ensuremath{\to} x = y\textsuperscript{2}}، \lr{dx/dy = 2y}.\PSbr{}\lr{f\textsubscript{Y}(y) = f\textsubscript{X}(y\textsuperscript{2})\ensuremath{\cdot}\textbar{}2y\textbar{} = e\textsuperscript{-y\textsuperscript{2}}\ensuremath{\cdot}2y} برای \lr{y \ensuremath{\ge} 0}. این یک توزیع رایلی با \lr{\ensuremath{\sigma}\textsuperscript{2} = 1/2} است.

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 4}

فاصله \lr{D \textasciitilde{} Uniform[1, 10] km}. تلفات مسیر: \lr{L = 20\ensuremath{\cdot}log\textsubscript{10}(D) dB}. مقدار \lr{E[L]} و \lr{PDF} مربوط به \lr{L} را بیابید.

\PSsolution{} \lr{E[L] = E[20\ensuremath{\cdot}log\textsubscript{10}(D)] = \ensuremath{\int}\textsubscript{1}\textsuperscript{10} 20\ensuremath{\cdot}log\textsubscript{10}(d)\ensuremath{\cdot}(1/9) dd = (20/9)\ensuremath{\cdot}\ensuremath{\int}\textsubscript{1}\textsuperscript{10} log\textsubscript{10}(d) dd}\PSbr{}\lr{= (20/9)\ensuremath{\cdot}[d\ensuremath{\cdot}log\textsubscript{10}(d) - d/ln(10)]\textsubscript{1}\textsuperscript{10} = (20/9)\ensuremath{\cdot}[10 - 10/ln10 + 1/ln10] = (20/9)\ensuremath{\cdot}(10 - 9/ln10) \ensuremath{\approx} 13.55 dB}

\lr{PDF: L} از 0 تا \lr{20 dB} تغییر می‌کند. \lr{g\textsuperscript{-1}(l) = 10\textsuperscript{l/20}}، \lr{\textbar{}dg\textsuperscript{-1}/dl\textbar{} = (ln10/20)\ensuremath{\cdot}10\textsuperscript{l/20}}.\PSbr{}\lr{f\textsubscript{L}(l) = (1/9)\ensuremath{\cdot}(ln10/20)\ensuremath{\cdot}10\textsuperscript{l/20}} برای \lr{0 \ensuremath{\le} l \ensuremath{\le} 20}.

\end{PSbox}
