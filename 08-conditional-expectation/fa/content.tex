\begin{PSbox}[unbreakable]{obj}{اهداف یادگیری}

پس از تکمیل این ماژول، دانشجویان قادر خواهند بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  امیدهای ریاضی و واریانس‌های شرطی را محاسبه کنند
\item
  قانون امید ریاضی کل را به کار ببرند
\item
  قانون واریانس کل را به کار ببرند
\item
  از این ابزارها برای تحلیل مهندسی در شرایط کاری متعدد استفاده کنند
\end{enumerate}

\end{PSbox}

\PSsection{8.1}{مقدمه}

بسیاری از سیستم‌های مهندسی تحت \textbf{شرایط مختلف} کار می‌کنند (وضعیت‌های کانال، حالت‌ها، محیط‌ها). ما اغلب به \textbf{عملکرد میانگین کل} نیاز داریم که از عملکرد شرطی در هر وضعیت محاسبه می‌شود.

\begin{itemize}
\tightlist
\item
  میانگین \lr{\mbox{BER}} چقدر است هنگامی که کانال به‌طور تصادفی میان خوب و بد جابه‌جا می‌شود؟
\item
  واریانس کل اندازه‌گیری چقدر است هنگامی که حسگر در محیط‌های مختلف کار می‌کند؟
\end{itemize}

\PSsection{8.2}{امید ریاضی شرطی \lr{$E[X \mid Y = y]$}}

\begin{PSbox}[unbreakable]{def}{تعریف}

\textbf{پیوسته:}\PSdisplay{E[X|Y=y] = \int_{-\infty}^{\infty} x \cdot f_{X|Y}(x|y) \, dx}

\textbf{گسسته:}\PSdisplay{E[X|Y=y] = \sum_x x \cdot p_{X|Y}(x|y)}

\end{PSbox}

\PSsubsection{تفسیر}

\lr{$E[X \mid Y = y]$} همان \textbf{مقدار میانگین \lr{$X$}} است هنگامی که \lr{$Y = y$} را می‌دانیم. این تابعی از \lr{$y$} است.

\PSsubsection{\lr{$E[X \mid Y]$} به‌عنوان یک متغیر تصادفی}

هنگامی که خود \lr{$Y$} تصادفی باشد، \lr{$E[X \mid Y]$} یک \textbf{متغیر تصادفی} است \lr{—} وقتی \lr{$Y = y$} باشد مقدار \lr{$E[X \mid Y = y]$} را اختیار می‌کند.

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی}

توان دریافتی میانگین به‌شرط وضعیت کانال:

\begin{itemize}
\tightlist
\item
  کانال خوب \lr{\mbox{$(Y = 1)$: $E[P \mid Y = 1] = 10$ mW}}
\item
  کانال ضعیف \lr{\mbox{$(Y = 0)$: $E[P \mid Y = 0] = 1$ mW}}
\item
  \lr{$P(Y = 1) = 0.7$}، \lr{$P(Y = 0) = 0.3$}
\end{itemize}

\end{PSbox}

\PSsection{8.3}{واریانس شرطی \lr{$\operatorname{Var}(X \mid Y = y)$}}

\begin{PSbox}[unbreakable]{def}{تعریف}

\PSdisplay{\PSmtext{Var}(X|Y=y) = E[(X - E[X|Y=y])^2 | Y=y] = E[X^2|Y=y] - (E[X|Y=y])^2}

\end{PSbox}

\PSsubsection{تفسیر}

واریانس شرطی \textbf{عدم‌قطعیت باقی‌مانده در \lr{$X$}} را پس از معلوم‌شدن \lr{$Y$} می‌سنجد.

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی}

دقت اندازه‌گیری به دما وابسته است:

\begin{itemize}
\tightlist
\item
  در \lr{$T = 20^{\circ}C$: $\operatorname{Var}(X \mid T = 20) = 0.01$} (دقت بالا)
\item
  در \lr{$T = 80^{\circ}C$: $\operatorname{Var}(X \mid T = 80) = 0.09$} (دقت پایین به‌دلیل اثرهای گرمایی)
\end{itemize}

\end{PSbox}

\PSsection{8.4}{قانون امید ریاضی کل (خاصیت برج)}

\PSsubsection{بیان}

\PSdisplay{E[X] = E[E[X|Y]]}

\textbf{نسخه گسسته:}\PSdisplay{E[X] = \sum_y E[X|Y=y] \cdot P(Y=y)}

\textbf{نسخه پیوسته:}\PSdisplay{E[X] = \int_{-\infty}^{\infty} E[X|Y=y] \cdot f_Y(y) \, dy}

\PSsubsection{اثبات (گسسته)}

\lr{$E[E[X \mid Y]] = \sum_{y} E[X \mid Y = y] \cdot P(Y = y)$}\PSbr{}\lr{$= \sum_{y} \left[\sum_{x} x \cdot P(X = x \mid Y = y)\right] \cdot P(Y = y)$}\PSbr{}\lr{$= \sum_{x} \Sigma_{y} x \cdot P(X = x,\, Y = y)$}\PSbr{}\lr{\mbox{$= \sum_{x} x \cdot P(X = x) = E[X]$ \PSyes{}}}

\PSsubsection{تفسیر}

\textbf{میانگین کل} برابر است با \textbf{میانگین وزنی میانگین‌های شرطی}، با وزن احتمال هر شرط.

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: میانگین \lr{\mbox{BER}} در وضعیت‌های کانال}

یک سیستم بی‌سیم میان سه حالت جابه‌جا می‌شود:

\begin{itemize}
\tightlist
\item
  حالت 1 (60\% زمان): \lr{$\mathrm{BER} = 10^{-6}$}
\item
  حالت 2 (30\% زمان): \lr{$\mathrm{BER} = 10^{-4}$}
\item
  حالت 3 (10\% زمان): \lr{$\mathrm{BER} = 10^{-2}$}
\end{itemize}

\lr{\mbox{BER}} کل \lr{$= E[\mathrm{BER}] = 0.6(10^{-6}) + 0.3(10^{-4}) + 0.1(10^{-2})$}\PSbr{}\lr{$= 0.6 \times 10^{-6} + 30 \times 10^{-6} + 10000 \times 10^{-6} \approx 1.003 \times 10^{-3}$}

\textbf{نکته مهندسی:} \lr{\mbox{BER}} کل تحت تسلط بدترین حالت است، هرچند که این حالت تنها در 10\% زمان رخ می‌دهد.

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: عمر میانگین با انواع نقص}

قطعات بدون نقص (90\%) یا دارای نقص پنهان (10\%) هستند:

\begin{itemize}
\tightlist
\item
  بدون نقص: \lr{$E[T \mid \PSmtext{good}] = 10000$} ساعت
\item
  با نقص: \lr{$E[T \mid \PSmtext{defect}] = 500$} ساعت
\end{itemize}

\lr{$E[T] = 0.9(10000) + 0.1(500) = 9050$} ساعت

\end{PSbox}

\PSsection{8.5}{قانون واریانس کل}

\PSsubsection{بیان}

\PSdisplay{\PSmtext{Var}(X) = E[\PSmtext{Var}(X|Y)] + \PSmtext{Var}(E[X|Y])}

\lr{$\operatorname{Var}(X)$} & واریانس کل & عدم‌قطعیت کل در \lr{$X$} \\

\end{longtable}
}

\PSsubsection{اثبات}

\lr{$\operatorname{Var}(X) = E[X^{2}] - (E[X])^{2}$}

\lr{$E[X^{2}] = E[E[X^{2} \mid Y]]$} (خاصیت برج)\PSbr{}\lr{$= E[\operatorname{Var}(X \mid Y) + (E[X \mid Y])^{2}]$}\PSbr{}\lr{$= E[\operatorname{Var}(X \mid Y)] + E[(E[X \mid Y])^{2}]$}

\lr{$(E[X])^{2} = (E[E[X \mid Y]])^{2}$}

\lr{$\operatorname{Var}(X) = E[\operatorname{Var}(X \mid Y)] + E[(E[X \mid Y])^{2}] - (E[E[X \mid Y]])^{2}$}\PSbr{}\lr{$= E[\operatorname{Var}(X \mid Y)] + \operatorname{Var}(E[X \mid Y])$ \PSyes{}}

\PSsubsection{تفسیر: تجزیه عدم‌قطعیت}

عدم‌قطعیت کل = میانگین عدم‌قطعیت درون هر شرط + تغییرپذیری میانگین میان شرایط

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: حسگر در دو محیط}

یک حسگر در فضای بسته (\lr{$Y = 1$}، احتمال \PSnum{0.6)} و فضای باز (\lr{$Y = 2$}، احتمال \PSnum{0.4)} کار می‌کند:

\begin{itemize}
\tightlist
\item
  فضای بسته: \lr{$E[X \mid Y = 1] = 25^{\circ}C$}، \lr{$\operatorname{Var}(X \mid Y = 1) = 1$}
\item
  فضای باز: \lr{$E[X \mid Y = 2] = 15^{\circ}C$}، \lr{$\operatorname{Var}(X \mid Y = 2) = 16$}
\end{itemize}

**\lr{$E[\operatorname{Var}(X \mid Y)] \ast \ast = 0.6(1) + 0.4(16) = 7.0$} (میانگین واریانس درون‌شرطی)

\lr{**$\operatorname{Var}(E[X \mid Y]) \ast \ast = E[(E[X \mid Y])^{2}] - (E[E[X \mid Y]])^{2}$}\PSbr{}\lr{$= 0.6(25^{2}) + 0.4(15^{2}) - [0.6(25) + 0.4(15)]^{2} = 375 + 90 - (21)^{2} = 465 - 441 = 24$}

\lr{\mbox{**$\operatorname{Var}(X) \ast \ast = 7.0 + 24.0 = 31.0$}}

بیشتر واریانس از تفاوت \textbf{بین‌شرطی} ناشی می‌شود (24 از 31).

\end{PSbox}

\PSsection{8.6}{مسایل تمرینی}

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

یک لینک دیجیتال روی دو نوع کانال کار می‌کند:

\begin{itemize}
\tightlist
\item
  نوع \lr{$A$} (احتمال \PSnum{0.7}): \lr{$E[\PSmtext{delay}\ \mid A] = 5\,\mathrm{ms}$}، \lr{$\operatorname{Var}(\PSmtext{delay}\ \mid A) = 2$}
\item
  نوع \lr{$B$} (احتمال \PSnum{0.3}): \lr{$E[\PSmtext{delay}\ \mid B] = 20\,\mathrm{ms}$}، \lr{$\operatorname{Var}(\PSmtext{delay}\ \mid B) = 25$}
\end{itemize}

موارد زیر را بیابید: (الف) \lr{$E[\PSmtext{delay}]$}، (ب) \lr{$\operatorname{Var}(\PSmtext{delay})$}.

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\PSalph{enumi})}
\tightlist
\item
  \lr{\mbox{$E[D] = 0.7(5) + 0.3(20) = 3.5 + 6 = 9.5$ ms}}
\item
  \lr{$E[\operatorname{Var}(D \mid \PSmtext{type})] = 0.7(2) + 0.3(25) = 1.4 + 7.5 = 8.9$}\PSbr{}\lr{$\operatorname{Var}(E[D \mid \PSmtext{type}]) = 0.7(5^{2}) + 0.3(20^{2}) - 9.5^{2} = 17.5 + 120 - 90.25 = 47.25$}\PSbr{}\lr{$\operatorname{Var}(D) = 8.9 + 47.25 = 56.15$}
\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2}

تعداد تصادفی \lr{$N$} از قطعات آزمایش می‌شوند. \lr{$N \sim \mathrm{Poisson}(10)$}. هر قطعه احتمال خرابی مستقل \lr{$p = 0.05$} دارد. فرض کنید \lr{$X$} = تعداد خرابی‌ها. \lr{$E[X]$} و \lr{$\operatorname{Var}(X)$} را بیابید.

\PSsolution{} \lr{$X \mid N \sim \mathrm{Binomial}(N,\, 0.05)$. $E[X \mid N] = 0.05\,\mathrm{N}$. $\operatorname{Var}(X \mid N) = N(0.05)(0.95) = 0.0475\,\mathrm{N}$.}\PSbr{}\lr{$E[X] = E[E[X \mid N]] = E[0.05\,\mathrm{N}] = 0.05(10) = 0.5$}\PSbr{}\lr{$\operatorname{Var}(X) = E[\operatorname{Var}(X \mid N)] + \operatorname{Var}(E[X \mid N]) = E[0.0475\,\mathrm{N}] + \operatorname{Var}(0.05\,\mathrm{N}) = 0.0475(10) + 0.05^{2}(10) = 0.475 + 0.025 = 0.5$}

\end{PSbox}
