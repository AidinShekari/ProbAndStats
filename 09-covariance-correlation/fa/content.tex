\begin{PSbox}[unbreakable]{obj}{اهداف یادگیری}

پس از تکمیل این ماژول، دانشجویان قادر خواهند بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  کوواریانس را محاسبه و تفسیر کنند
\item
  ضریب همبستگی را محاسبه و تفسیر کنند
\item
  میان استقلال و ناهمبستگی تمایز قایل شوند
\item
  ماتریس‌های همبستگی را بسازند و تفسیر کنند
\item
  تحلیل همبستگی را در مسایل مهندسی به کار ببرند
\end{enumerate}

\end{PSbox}

\PSsection{9.1}{مقدمه: اندازه‌گیری روابط خطی}

هنگامی که دو متغیر تصادفی مستقل نباشند، باید \textbf{نحوه ارتباط آن‌ها} را کمّی کنیم. کوواریانس و همبستگی \textbf{رابطه خطی} میان دو متغیر تصادفی را می‌سنجند.

مثال‌های مهندسی:

\begin{itemize}
\tightlist
\item
  نویز همبسته در کانال‌های حسگر مجاور
\item
  اندازه‌گیری سیگنال در دو آنتن (همبستگی \lr{MIMO})
\item
  ولتاژ و جریان در یک مدار (مرتبط از طریق امپدانس)
\item
  دما و عملکرد قطعه (وابستگی گرمایی)
\end{itemize}

\PSsection{9.2}{کوواریانس}

\begin{PSbox}[unbreakable]{def}{تعریف}

\PSdisplay{\PSmtext{Cov}(X,Y) = E[(X - \mu_X)(Y - \mu_Y)] = E[XY] - E[X]E[Y]}

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \lr{Cov(X, X) = Var(X)}
\item
  \lr{Cov(X, Y) = Cov(Y, X)} (متقارن)
\item
  \lr{Cov(aX + b, cY + d) = ac \ensuremath{\cdot} Cov(X, Y)}
\item
  اگر \lr{X}، \lr{Y} مستقل باشند \lr{\ensuremath{\to} Cov(X, Y) = 0}
\item
  \lr{Var(X + Y) = Var(X) + Var(Y) + 2Cov(X, Y)}
\item
  \lr{Var(X - Y) = Var(X) + Var(Y) - 2Cov(X, Y)}
\end{enumerate}

\PSsubsection{تفسیر علامت}

{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[c]{@{}cc@{}}
\toprule\noalign{}
\rowcolor{PSnavy}\PSth{\lr{Cov(X,Y)}} & \PSth{معنا} \\
\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
\lr{\textgreater{}} \lr{0} & \lr{X} و \lr{Y} تمایل دارند با هم افزایش یابند \\
\lr{\textless{}} \lr{0} & وقتی \lr{X} افزایش می‌یابد، \lr{Y} تمایل به کاهش دارد \\
= \lr{0} & رابطه خطی وجود ندارد (ناهمبسته) \\
\end{longtable}
}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: بهره تقویت‌کننده و خروجی}

اگر بهره \lr{G} به‌طور تصادفی تغییر کند و ورودی در \lr{V\textsubscript{in}} ثابت باشد:

\begin{itemize}
\tightlist
\item
  خروجی: \lr{V\textsubscript{out} = G \ensuremath{\cdot} V\textsubscript{in}}
\item
  \lr{Cov(G, V\textsubscript{out}) = Cov(G, G\ensuremath{\cdot}V\textsubscript{in}) = V\textsubscript{in} \ensuremath{\cdot} Var(G) \textgreater{} 0}
\end{itemize}

بهره بالاتر \lr{\ensuremath{\to}} خروجی بالاتر (کوواریانس مثبت).

\end{PSbox}

\PSsection{9.3}{ضریب همبستگی}

\begin{PSbox}[unbreakable]{def}{تعریف}

\PSdisplay{\rho_{XY} = \frac{\PSmtext{Cov}(X,Y)}{\sigma_X \sigma_Y}}

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{کران‌دار:} \lr{-1 \ensuremath{\le} \ensuremath{\rho} \ensuremath{\le} 1}
\item
  \lr{\textbf{\lr{\ensuremath{\rho} = 1}:}} رابطه خطی مثبت کامل (\lr{Y = aX + b}، \lr{a \textgreater{} 0})
\item
  \lr{\textbf{\lr{\ensuremath{\rho} = -1}:}} رابطه خطی منفی کامل (\lr{Y = aX + b}، \lr{a \textless{} 0})
\item
  \lr{\textbf{\lr{\ensuremath{\rho} = 0}:}} ناهمبسته (بدون رابطه خطی)
\item
  \textbf{بی‌بُعد:} برخلاف کوواریانس، \lr{\ensuremath{\rho}} به یکاها وابسته نیست
\end{enumerate}

\PSsubsection{مقیاس تفسیر}

{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[c]{@{}cc@{}}
\toprule\noalign{}
\rowcolor{PSnavy}\PSth{بازه} & \PSth{\lr{\ensuremath{\rho}}} \\
\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
\lr{0.0 – 0.2} & بسیار ضعیف/ناچیز \\
\lr{0.2 – 0.4} & ضعیف \\
\lr{0.4 – 0.6} & متوسط \\
\lr{0.6 – 0.8} & قوی \\
\lr{0.8 – 1.0} & بسیار قوی \\
\end{longtable}
}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: محوشدگی همبسته}

دو مسیر آنتنی با همبستگی \lr{\ensuremath{\rho} = 0.3}:

\begin{itemize}
\tightlist
\item
  همبستگی کم \lr{\ensuremath{\to}} تنوع به‌خوبی کار می‌کند (محوشدگی مستقل)
\item
  همبستگی زیاد \lr{\ensuremath{\to}} بهره تنوع کاهش می‌یابد
\end{itemize}

طراحی سیستم \lr{MIMO} نیازمند: \lr{\ensuremath{\rho} \textless{} 0.5} برای مالتی‌پلکسینگ فضایی مؤثر.

\end{PSbox}

\PSsection{9.4}{همبستگی متقابل}

\begin{PSbox}[unbreakable]{def}{تعریف}

\PSdisplay{R_{XY} = E[XY]}

\end{PSbox}

\PSsubsection{رابطه با کوواریانس}

\PSdisplay{\PSmtext{Cov}(X,Y) = R_{XY} - \mu_X \mu_Y}

اگر \lr{X} یا \lr{Y} میانگین صفر داشته باشد: \lr{Cov(X,Y) = E[XY] = R\textsubscript{XY}}

\PSsubsection{اهمیت مهندسی}

همبستگی متقابل در موارد زیر بنیادی است:

\begin{itemize}
\tightlist
\item
  فیلترهای منطبق (گیرنده‌های همبستگی در مخابرات)
\item
  پردازش سیگنال رادار (تشخیص تأخیر زمانی)
\item
  آشکارسازی سیگنال (همبسته‌سازی دریافتی با الگوی معلوم)
\end{itemize}

\PSsection{9.5}{استقلال در برابر ناهمبستگی}

\PSsubsection{تمایز حیاتی}

{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[c]{@{}cc@{}}
\toprule\noalign{}
\rowcolor{PSnavy}\PSth{گزاره} & \PSth{همیشه درست است؟} \\
\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
مستقل \lr{\ensuremath{\to}} ناهمبسته & \lr{\PSyes{}} بله \\
ناهمبسته \lr{\ensuremath{\to}} مستقل & \lr{\PSno{}} خیر (به‌طور کلی) \\
ناهمبسته \lr{\ensuremath{\to}} مستقل (گاوسی مشترک) & \lr{\PSyes{}} بله \\
\end{longtable}
}

\PSsubsection{چرا مستقل \lr{\ensuremath{\to}} ناهمبسته}

اگر \lr{X}، \lr{Y} مستقل باشند: \lr{E[XY] = E[X]E[Y]}، بنابراین \lr{Cov(X,Y) = E[XY] - E[X]E[Y] = 0}.

\begin{PSbox}[unbreakable]{ex}{مثال نقض: ناهمبسته اما وابسته}

فرض کنید \lr{X \textasciitilde{} N(0,1)} و \lr{Y = X\textsuperscript{2}}. آنگاه:

\begin{itemize}
\tightlist
\item
  \lr{Cov(X, Y) = E[XY] - E[X]E[Y] = E[X\textsuperscript{3}] - 0 = 0} (چون \lr{X} متقارن است)
\item
  اما \lr{Y} کاملاً توسط \lr{X} تعیین می‌شود \lr{\ensuremath{\to}} وابستگی حداکثری!
\end{itemize}

\textbf{\lr{\ensuremath{\rho} = 0} به معنای استقلال نیست} (به‌جز برای متغیرهای تصادفی گاوسی مشترک).

\end{PSbox}

\PSsubsection{استثنای گاوسی}

برای متغیرهای تصادفی \textbf{گاوسی مشترک}:

\PSdisplay{\PSmtext{Uncorrelated} \Leftrightarrow \PSmtext{Independent}}

به همین دلیل توزیع گاوسی در مهندسی بسیار راحت است \lr{—} بررسی \lr{\ensuremath{\rho} = 0} برای اثبات استقلال کافی است.

\PSsubsection{پیام مهندسی}

در مخابرات با نویز گاوسی:

\begin{itemize}
\tightlist
\item
  اگر نمونه‌های نویز ناهمبسته باشند \lr{\ensuremath{\to}} مستقل هستند
\item
  این موضوع تحلیل را به‌شدت ساده می‌کند
\item
  برای تداخل غیرگاوسی: ناهمبستگی استقلال را تضمین نمی‌کند
\end{itemize}

\PSsection{9.6}{ماتریس همبستگی}

\begin{PSbox}[unbreakable]{def}{تعریف برای بردار تصادفی \lr{X = [X\textsubscript{1}, X\textsubscript{2}, …, X\textsubscript{n}]\textsuperscript{T}}}

\textbf{ماتریس کوواریانس} \lr{\ensuremath{\Sigma}}:

\PSdisplay{\Sigma_{ij} = \PSmtext{Cov}(X_i, X_j)}

\PSdisplay{\Sigma = \begin{bmatrix} \PSmtext{Var}(X_1) & \PSmtext{Cov}(X_1,X_2) & \cdots \\ \PSmtext{Cov}(X_2,X_1) & \PSmtext{Var}(X_2) & \cdots \\ \vdots & & \ddots \end{bmatrix}}

\textbf{ماتریس همبستگی} \lr{R} (نرمال‌شده):

\PSdisplay{R_{ij} = \rho_{X_i X_j} = \frac{\Sigma_{ij}}{\sqrt{\Sigma_{ii}\Sigma_{jj}}}}

\end{PSbox}

\PSsubsection{ویژگی‌های ماتریس کوواریانس}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  متقارن: \lr{\ensuremath{\Sigma} = \ensuremath{\Sigma}\textsuperscript{T}}
\item
  نیم‌معین مثبت: \lr{x\textsuperscript{T}\ensuremath{\Sigma}x \ensuremath{\ge} 0} برای تمام \lr{x}
\item
  درایه‌های قطری = واریانس‌ها
\item
  درایه‌های غیرقطری = کوواریانس‌ها
\end{enumerate}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: سه حسگر}

سه حسگر دما با:

\begin{itemize}
\tightlist
\item
  \lr{\ensuremath{\sigma}\textsubscript{1} = \ensuremath{\sigma}\textsubscript{2} = \ensuremath{\sigma}\textsubscript{3} = 1\ensuremath{^{\circ}}C}
\item
  \lr{\ensuremath{\rho}\textsubscript{12} = 0.8} (حسگرهای 1 و 2 نزدیک هم هستند)
\item
  \lr{\ensuremath{\rho}\textsubscript{13} = 0.3} (حسگر 3 دورتر است)
\item
  \lr{\ensuremath{\rho}\textsubscript{23} = 0.4}
\end{itemize}

ماتریس همبستگی:

\tcbset{PScodemode/.style={unbreakable}}

\begin{PSplaincode}
\begin{Verbatim}
R = [1.0  0.8  0.3]
    [0.8  1.0  0.4]
    [0.3  0.4  1.0]
\end{Verbatim}
\end{PSplaincode}

\end{PSbox}

\PSsection{9.7}{واریانس ترکیب‌های خطی}

\PSsubsection{فرمول کلی}

\PSdisplay{\PSmtext{Var}\left(\sum_{i=1}^n a_i X_i\right) = \sum_{i=1}^n a_i^2 \PSmtext{Var}(X_i) + 2\sum_{i<j} a_i a_j \PSmtext{Cov}(X_i, X_j)}

در شکل ماتریسی: \lr{Var(a\textsuperscript{T}X) = a\textsuperscript{T}\ensuremath{\Sigma}a}

\PSsubsection{حالت خاص: جمع دو متغیر تصادفی}

\lr{Var(X + Y) = Var(X) + Var(Y) + 2Cov(X,Y) = \ensuremath{\sigma}\textsubscript{X}\textsuperscript{2} + \ensuremath{\sigma}\textsubscript{Y}\textsuperscript{2} + 2\ensuremath{\rho}\ensuremath{\sigma}\textsubscript{X\ensuremath{\sigma}}\textsubscript{Y}}

{\def\LTcaptype{none} % do not increment counter
\begin{longtable}[c]{@{}ccc@{}}
\toprule\noalign{}
\rowcolor{PSnavy}\PSth{همبستگی} & \PSth{\lr{Var(X+Y)}} & \PSth{اثر} \\
\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
\lr{\ensuremath{\rho} = 1} & \lr{(\ensuremath{\sigma}\textsubscript{X} + \ensuremath{\sigma}\textsubscript{Y})\textsuperscript{2}} & بیشینه \lr{—} واریانس‌ها به‌صورت سازنده جمع می‌شوند \\
\lr{\ensuremath{\rho} = 0} & \lr{\ensuremath{\sigma}\textsubscript{X}\textsuperscript{2} + \ensuremath{\sigma}\textsubscript{Y}\textsuperscript{2}} & جمع مستقل \\
\lr{\ensuremath{\rho} = -1} & \lr{(\ensuremath{\sigma}\textsubscript{X} - \ensuremath{\sigma}\textsubscript{Y})\textsuperscript{2}} & کمینه \lr{—} حذف متقابل \\
\end{longtable}
}

\PSsubsection{کاربرد مهندسی: میانگین‌گیری از حسگرهای همبسته}

اگر \lr{n} حسگر واریانس برابر \lr{\ensuremath{\sigma}\textsuperscript{2}} و همبستگی زوجی \lr{\ensuremath{\rho}} داشته باشند:

\PSdisplay{\PSmtext{Var}(\bar{X}) = \frac{\sigma^2}{n}[1 + (n-1)\rho]}

\begin{itemize}
\tightlist
\item
  اگر \lr{\ensuremath{\rho} = 0} (مستقل): \lr{Var(\ensuremath{\bar{X}}) = \ensuremath{\sigma}\textsuperscript{2}/n} (بهره کامل میانگین‌گیری)
\item
  اگر \lr{\ensuremath{\rho} = 1} (یکسان): \lr{Var(\ensuremath{\bar{X}}) = \ensuremath{\sigma}\textsuperscript{2}} (بدون بهره از میانگین‌گیری!)
\item
  اگر \lr{\ensuremath{\rho} = 0.5}، \lr{n = 4: Var(\ensuremath{\bar{X}}) = \ensuremath{\sigma}\textsuperscript{2}(1 + 3\ensuremath{\times}0.5)/4 = 0.625\ensuremath{\sigma}\textsuperscript{2}} (بهره کاهش‌یافته)
\end{itemize}

\PSsection{9.9}{مسایل تمرینی}

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

برای \lr{X} و \lr{Y} داریم: \lr{E[X]=2}، \lr{E[Y]=3}، \lr{Var(X)=4}، \lr{Var(Y)=9}، \lr{E[XY]=8}.

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  \lr{Cov(X,Y)} را بیابید. \lr{(b) \ensuremath{\rho}} را بیابید. \lr{(c) Var(X+Y)} را بیابید. \lr{(d) Var(2X-3Y)} را بیابید.
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  \lr{Cov(X,Y) = E[XY] - E[X]E[Y] = 8 - 2(3) = 2}
\item
  \lr{\ensuremath{\rho} = 2/(\ensuremath{\surd}4\ensuremath{\cdot}\ensuremath{\surd}9) = 2/6 = 1/3}
\item
  \lr{Var(X+Y) = 4 + 9 + 2(2) = 17}
\item
  \lr{Var(2X-3Y) = 4(4) + 9(9) + 2(2)(-3)(2) = 16 + 81 - 24 = 73}
\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2}

چهار حسگر یکسان \lr{(\ensuremath{\sigma}\textsuperscript{2}=1)} همبستگی زوجی \lr{\ensuremath{\rho}=0.5} دارند. \lr{Var(\ensuremath{\bar{X}})} چقدر است؟

\PSsolution{} \lr{Var(\ensuremath{\bar{X}}) = \ensuremath{\sigma}\textsuperscript{2}/n \ensuremath{\cdot} [1 + (n-1)\ensuremath{\rho}] = 1/4 \ensuremath{\cdot} [1 + 3(0.5)] = 0.25 \ensuremath{\times} 2.5 = 0.625}

مقایسه کنید: اگر مستقل باشند \lr{(\ensuremath{\rho}=0): Var(\ensuremath{\bar{X}}) = 0.25}. همبستگی اثربخشی میانگین‌گیری را کاهش می‌دهد.

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 3}

با داده \lr{X \textasciitilde{} N(0,1)}، تعریف کنید \lr{Y = X} هنگامی که \lr{\textbar{}}\lr{X\textbar{}\textless{}1} و \lr{Y = -X} هنگامی که \lr{\textbar{}}\lr{X\textbar{}\ensuremath{\ge}1}. نشان دهید در این حالت \lr{Cov(X,Y) \ensuremath{\neq} 0} اما \lr{E[X\ensuremath{\cdot}Y]} قابل محاسبه است.

\PSsolution{} \lr{E[XY] = E[X\textsuperscript{2}\ensuremath{\cdot}1(\textbar{}X\textbar{}\textless{}1)] + E[-X\textsuperscript{2}\ensuremath{\cdot}1(\textbar{}X\textbar{}\ensuremath{\ge}1)] = E[X\textsuperscript{2}\ensuremath{\cdot}1(\textbar{}X\textbar{}\textless{}1)] - E[X\textsuperscript{2}\ensuremath{\cdot}1(\textbar{}X\textbar{}\ensuremath{\ge}1)]}\PSbr{}= \lr{P(\textbar{}X\textbar{}\textless{}1)E[X\textsuperscript{2}\textbar{} \textbar{}X\textbar{}\textless{}1] - P(\textbar{}X\textbar{}\ensuremath{\ge}1)E[X\textsuperscript{2}\textbar{} \textbar{}X\textbar{}\ensuremath{\ge}1]}\PSbr{}این مقدار ناصفر است زیرا \lr{E[X\textsuperscript{2}\textbar{} \textbar{}X\textbar{}\textless{}1] \ensuremath{\neq} E[X\textsuperscript{2}\textbar{} \textbar{}X\textbar{}\ensuremath{\ge}1]}، بنابراین \lr{Cov(X,Y) = E[XY] - 0 \ensuremath{\neq} 0}.\PSbr{}\lr{X} و \lr{Y} به‌روشنی وابسته هستند (\lr{Y} از \lr{X} تعریف شده است) و در این حالت همبسته نیز هستند.
