\begin{PSbox}[unbreakable]{obj}{اهداف یادگیری}

پس از تکمیل این ماژول، دانشجویان قادر خواهند بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  توزیع‌های مشترک را به \lr{$N$} متغیر تصادفی تعمیم دهند
\item
  بردارهای تصادفی را تعریف کنند و با آن‌ها کار کنند
\item
  بردار میانگین و ماتریس‌های کوواریانس را محاسبه کنند
\item
  توزیع گاوسی چندمتغیره را به کار ببرند
\item
  تبدیل‌های خطی بردارهای تصادفی را انجام دهند
\end{enumerate}

\end{PSbox}

\PSsection{10.1}{مقدمه: از 2 به \lr{$N$} متغیر}

سیستم‌های واقعی اغلب شامل چندین کمیت نامعین به‌طور همزمان هستند:

\begin{itemize}
\tightlist
\item
  \textbf{آرایه حسگر:} \lr{$N$} حسگر که هر یک یک اندازه‌گیری نویزی تولید می‌کند
\item
  \textbf{سیستم \lr{\mbox{MIMO}}:} ضرایب کانال \lr{$M \times N$}
\item
  \textbf{برآورد پارامتر:} \lr{$p$} پارامتر نامعلوم
\item
  \textbf{نمونه‌های سیگنال:} \lr{$N$} نمونه نویز متوالی
\end{itemize}

به چارچوبی برای بردارهای تصادفی \textbf{\lr{\mbox{N}}-بعدی} نیاز داریم.

\PSsection{10.2}{توزیع‌های مشترک برای \lr{$N$} متغیر تصادفی}

\PSsubsection{\lr{\mbox{PDF}} مشترک}

برای \lr{$X_{1},\, X_{2},\, \ldots,\, X_{n}$}:\PSdisplay{f_{X_1,...,X_n}(x_1,...,x_n) \geq 0, \quad \int\cdots\int f_{X_1,...,X_n} \, dx_1 \cdots dx_n = 1}

\PSsubsection{توزیع‌های حاشیه‌ای}

متغیرهای ناخواسته را انتگرال بگیرید (یا جمع کنید):\PSdisplay{f_{X_1}(x_1) = \int\cdots\int f_{X_1,...,X_n}(x_1,...,x_n) \, dx_2 \cdots dx_n}

\PSsubsection{توزیع‌های شرطی}

\PSdisplay{f_{X_1|X_2,...,X_n}(x_1|x_2,...,x_n) = \frac{f_{X_1,...,X_n}(x_1,...,x_n)}{f_{X_2,...,X_n}(x_2,...,x_n)}}

\PSsubsection{استقلال متقابل}

\lr{$X_{1},\, \ldots,\, X_{n}$} به‌طور متقابل مستقل هستند اگر:\PSdisplay{f_{X_1,...,X_n}(x_1,...,x_n) = \prod_{i=1}^n f_{X_i}(x_i)}

\PSsection{10.3}{بردارهای تصادفی}

\begin{PSbox}[unbreakable]{def}{تعریف}

یک \textbf{بردار تصادفی} مجموعه‌ای مرتب از متغیرهای تصادفی است:

\PSdisplay{\mathbf{X} = \begin{bmatrix} X_1 \\ X_2 \\ \vdots \\ X_n \end{bmatrix}}

\end{PSbox}

\PSsection{10.4}{بردار میانگین}

\begin{PSbox}[unbreakable]{def}{تعریف}

\PSdisplay{\boldsymbol{\mu} = E[\mathbf{X}] = \begin{bmatrix} E[X_1] \\ E[X_2] \\ \vdots \\ E[X_n] \end{bmatrix} = \begin{bmatrix} \mu_1 \\ \mu_2 \\ \vdots \\ \mu_n \end{bmatrix}}

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{itemize}
\tightlist
\item
  \lr{\mbox{$E[AX + b]$ = AE[X] $+ b = A\mu + b$}}
\item
  خطی‌بودن به بردارها و ماتریس‌ها تعمیم می‌یابد
\end{itemize}

\PSsection{10.5}{ماتریس کوواریانس}

\begin{PSbox}[unbreakable]{def}{تعریف}

\PSdisplay{\boldsymbol{\Sigma} = E[(\mathbf{X} - \boldsymbol{\mu})(\mathbf{X} - \boldsymbol{\mu})^T]}

درایه \lr{$(i,\,j)$: $\Sigma_{ij} = \operatorname{Cov}(X_{i},\, X_{j})$}

\PSdisplay{\boldsymbol{\Sigma} = \begin{bmatrix} \sigma_1^2 & \PSmtext{Cov}(X_1,X_2) & \cdots & \PSmtext{Cov}(X_1,X_n) \\ \PSmtext{Cov}(X_2,X_1) & \sigma_2^2 & \cdots & \PSmtext{Cov}(X_2,X_n) \\ \vdots & & \ddots & \vdots \\ \PSmtext{Cov}(X_n,X_1) & \cdots & \cdots & \sigma_n^2 \end{bmatrix}}

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{متقارن:} \lr{$\Sigma = \Sigma^{T}$}
\item
  \textbf{نیم‌معین مثبت:} \lr{$v^{T}\Sigma v \ge 0$} برای تمام \lr{$v$}
\item
  \textbf{قطر = واریانس‌ها:} \lr{$\Sigma_{ii} = \operatorname{Var}(X_{i})$}
\item
  \textbf{غیرقطری = کوواریانس‌ها:} \lr{$\Sigma_{ij} = \operatorname{Cov}(X_{i},\, X_{j})$}
\item
  \textbf{مؤلفه‌های مستقل \lr{$\to \Sigma$} قطری}
\end{enumerate}

\PSsubsection{فرمول جایگزین}

\PSdisplay{\boldsymbol{\Sigma} = E[\mathbf{X}\mathbf{X}^T] - \boldsymbol{\mu}\boldsymbol{\mu}^T}

\PSsection{10.6}{توزیع گاوسی چندمتغیره}

\begin{PSbox}[unbreakable]{def}{تعریف}

\lr{$X \sim N(\mu,\, \Sigma)$} دارای \lr{\mbox{PDF}} زیر است:

\PSdisplay{f_\mathbf{X}(\mathbf{x}) = \frac{1}{(2\pi)^{n/2}|\boldsymbol{\Sigma}|^{1/2}} \exp\left(-\frac{1}{2}(\mathbf{x}-\boldsymbol{\mu})^T \boldsymbol{\Sigma}^{-1}(\mathbf{x}-\boldsymbol{\mu})\right)}

که در آن \lr{$\lvert \Sigma\rvert$} = دترمینان \lr{$\Sigma$}.

\end{PSbox}

\PSsubsection{ویژگی‌های کلیدی}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{حاشیه‌ای‌ها گاوسی هستند:} هر زیرمجموعه‌ای از مؤلفه‌ها به‌طور مشترک گاوسی است
\item
  \textbf{شرطی‌ها گاوسی هستند:} \lr{$X_{1} \mid X_{2} = x_{2}$} گاوسی است
\item
  \textbf{تبدیل‌های خطی:} \lr{\mbox{$Y$ = AX $+ b \sim N(A\mu +b,\, A\Sigma A^{T})$}}
\item
  \textbf{ناهمبسته = مستقل} (ویژه گاوسی!)
\item
  \textbf{کاملاً مشخص‌شده} با \lr{$\mu$} و \lr{$\Sigma$} (فقط 2 پارامتر برای هر متغیر به‌علاوه همبستگی‌ها لازم است)
\end{enumerate}

\PSsubsection{چرا این‌قدر مهم است؟}

\begin{itemize}
\tightlist
\item
  قضیه حد مرکزی: جمع‌های بسیاری از متغیرهای تصادفی \lr{\mbox{\ensuremath{\to}}} گاوسی چندمتغیره
\item
  بردارهای نویز حرارتی گاوسی چندمتغیره هستند
\item
  از نظر تحلیلی خوش‌رفتار است (حاشیه‌ای‌ها و شرطی‌های فرم بسته)
\item
  فیلترهای بهینه (کالمن، وینر) گاوسی‌بودن را فرض می‌کنند
\end{itemize}

\PSsubsection{حالت دوبعدی (گاوسی دوبعدی)}

برای \lr{$n = 2$} با \lr{$\mu = [0,\,0]^{T}$}، \lr{$\sigma_{1} = \sigma_{2} = 1$}:

\PSdisplay{f(x,y) = \frac{1}{2\pi\sqrt{1-\rho^2}} \exp\left(-\frac{x^2 - 2\rho xy + y^2}{2(1-\rho^2)}\right)}

خطوط تراز \textbf{بیضی} هستند (در حالت \lr{$\rho \ne 0$} کج شده‌اند).

\PSsection{10.7}{تبدیل‌های خطی بردارهای تصادفی}

\PSsubsection{تبدیل \lr{\mbox{$Y$ = AX $+ b$}}}

اگر \lr{$X$} یک بردار تصادفی با میانگین \lr{$\mu_{X}$} و کوواریانس \lr{$\Sigma_{X}$} باشد:

\PSdisplay{E[\mathbf{Y}] = A\boldsymbol{\mu}_X + \mathbf{b}}\PSdisplay{\boldsymbol{\Sigma}_Y = A\boldsymbol{\Sigma}_X A^T}

\PSsubsection{کاربردهای مهندسی}

\textbf{شکل‌دهی پرتو \lr{\mbox{(Beamforming)}}:} خروجی \lr{$y = w^{T}X$} (بردار وزن اعمال‌شده به آرایه حسگر)

\begin{itemize}
\tightlist
\item
  \lr{$E[y] = w^{T}\mu$}
\item
  \lr{$\operatorname{Var}(y) = w^{T}\Sigma w$}
\end{itemize}

\textbf{سفیدسازی:} ماتریس \lr{$W$} را بیابید که در آن \lr{$W\Sigma W^{T} = I$} (داده را ناهمبسته می‌کند)

\begin{itemize}
\tightlist
\item
  \lr{$W = \Sigma^{-1/2}$}
\end{itemize}

\textbf{تحلیل مؤلفه‌های اصلی:} چرخش برای قطری‌کردن \lr{$\Sigma$}

\PSsection{10.8}{مسایل تمرینی}

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

بردار تصادفی \lr{$X = [X_{1},\, X_{2},\, X_{3}]^{T}$} با \lr{$\mu = [1,\, 0,\, -1]^{T}$} و ماتریس کوواریانس \lr{$\Sigma = \begin{bmatrix} 4 & 1 & 0 \\ 1 & 2 & -1 \\ 0 & -1 & 3 \end{bmatrix}$}.

\begin{enumerate}
\def\labelenumi{(\PSalph{enumi})}
\tightlist
\item
  \lr{$\operatorname{Var}(X_{1} + X_{2} + X_{3})$} را بیابید. (ب) \lr{$\operatorname{Cov}(X_{1},\, X_{2}+X_{3})$} را بیابید. (ج) آیا \lr{$X_{1}$} و \lr{$X_{3}$} ناهمبسته هستند؟
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\PSalph{enumi})}
\tightlist
\item
  \lr{$\operatorname{Var}(X_{1}+X_{2}+X_{3}) = 1^{T}\Sigma 1 = 4+2+3 + 2(1) + 2(0) + 2(-1) = 9 + 0 = 9$}\PSbr{}توجه: \lr{$= \sum_{ij}$} تمام درایه‌ها \lr{$= 4+1+0+1+2+(-1)+0+(-1)+3 = 9$}
\item
  \lr{$\operatorname{Cov}(X_{1},\, X_{2}+X_{3}) = \operatorname{Cov}(X_{1},\,X_{2}) + \operatorname{Cov}(X_{1},\,X_{3}) = 1 + 0 = 1$}
\item
  \lr{$\operatorname{Cov}(X_{1},\,X_{3}) = 0$}، بنابراین بله، \lr{$X_{1}$} و \lr{$X_{3}$} ناهمبسته هستند.
\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2}

\lr{$X \sim N(0,\, \Sigma)$} با \lr{$\Sigma = \begin{bmatrix} 1 & \rho \\ \rho & 1 \end{bmatrix}$}. توزیع شرطی \lr{$X_{1} \mid X_{2} = x_{2}$} را بیابید.

\PSsolution{} برای گاوسی دوبعدی:\PSbr{}\lr{$\displaystyle X_{1} \mid X_{2} = x_{2} \sim N\left(\mu_{1} + \rho \left(\frac{\sigma_{1}}{\sigma_{2}}\right)(x_{2}-\mu_{2}),\, \sigma_{1}^{2}(1-\rho^{2})\right)$}\PSbr{}\lr{$= N(\rho x_{2},\, 1-\rho^{2})$}

میانگین شرطی تابعی خطی از \lr{$x_{2}$} است و واریانس شرطی با ضریب \lr{$(1-\rho^{2})$} کاهش می‌یابد.

\end{PSbox}
