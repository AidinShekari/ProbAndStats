\PSsection{2.1}{مقدمه: چرا متغیرهای تصادفی؟}

\begin{PSbox}[unbreakable]{prob}{مسیله}

در ماژول 1 با رویدادها کار کردیم \lr{—} زیرمجموعه‌هایی از یک فضای نمونه. اما تحلیل مهندسی به \textbf{اعداد} نیاز دارد:

\PSsection{2.2}{تعریف متغیر تصادفی}

\begin{PSbox}[unbreakable]{def}{تعریف صوری}

یک \textbf{متغیر تصادفی} \lr{X} تابعی است که هر خروجی \lr{\ensuremath{\omega}} در فضای نمونه \lr{S} را به یک عدد حقیقی نگاشت می‌کند:

\PSdisplay{X: S \rightarrow \mathbb{R}}\PSdisplay{\omega \mapsto X(\omega)}

\end{PSbox}

\PSsubsection{تفسیر}

متغیر تصادفی به هر خروجی ممکن آزمایش یک مقدار عددی نسبت می‌دهد. تصادفی‌بودن از این ناشی می‌شود که کدام خروجی \lr{\ensuremath{\omega}} رخ می‌دهد \lr{—} به‌محض تعیین \lr{\ensuremath{\omega}}، مقدار \lr{X(\ensuremath{\omega})} یک عدد معیّن است.

\PSsubsection{مثال مهندسی}

\textbf{آزمایش:} ارسال 5 بیت روی یک کانال نویزی.\PSbr{}\textbf{فضای نمونه:} \lr{S} = تمام دنباله‌های ممکن درست/خطا برای 5 بیت (\lr{2\textsuperscript{5} = 32} خروجی)\PSbr{}\textbf{متغیر تصادفی:} \lr{X} = تعداد خطاهای بیت

\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
\lr{(C,C,C,C,C)} & 0 \\

\PSsection{2.3}{متغیرهای تصادفی گسسته}

\begin{PSbox}[unbreakable]{def}{تعریف}

متغیر تصادفی \lr{X} \textbf{گسسته} است اگر مقادیر خود را از یک مجموعه \textbf{شمارا} (متناهی یا شمارای نامتناهی) بگیرد.

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{itemize}
\tightlist
\item
  مقادیر را می‌توان فهرست کرد: \lr{\{x\textsubscript{1}, x\textsubscript{2}, x\textsubscript{3}, …\}}
\item
  هر مقدار یک احتمال مشخص دارد (یک «جرم» از احتمال)
\item
  مجموع تمام احتمال‌ها = \lr{1}
\end{itemize}

\PSsection{2.4}{متغیرهای تصادفی پیوسته}

\begin{PSbox}[unbreakable]{def}{تعریف}

متغیر تصادفی \lr{X} \textbf{پیوسته} است اگر مقادیر خود را از یک مجموعه \textbf{ناشمارا} (بازه‌ای از اعداد حقیقی) بگیرد.

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{itemize}
\tightlist
\item
  \lr{X} می‌تواند هر مقدار در یک محدوده را اختیار کند (مثلاً \lr{[0, \ensuremath{\infty})} یا \lr{(-\ensuremath{\infty}, \ensuremath{\infty})})
\item
  احتمال هر مقدار دقیق منفرد صفر است: \lr{P(X = x) = 0}
\item
  احتمال‌ها برای بازه‌ها تعریف می‌شوند: \lr{P(a \ensuremath{\le} X \ensuremath{\le} b)}
\end{itemize}

\PSsection{2.5}{تابع جرم احتمال \lr{(PMF)}}

\begin{PSbox}[unbreakable]{def}{تعریف}

برای یک متغیر تصادفی گسسته \lr{X}، \textbf{تابع جرم احتمال} عبارت است از:

\PSdisplay{p_X(x) = P(X = x)}

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{نامنفی بودن:} \lr{p\textsubscript{X}(x) \ensuremath{\ge} 0} برای تمام \lr{x}
\item
  \textbf{نرمال‌سازی:} \lr{\ensuremath{\Sigma} p\textsubscript{X}(x\textsubscript{i}) = 1} (جمع روی تمام مقادیر ممکن)
\item
  \textbf{محاسبه احتمال:} \lr{P(X \ensuremath{\in} A) = \ensuremath{\Sigma}\textsubscript{x\ensuremath{\in}A} p\textsubscript{X}(x)}
\end{enumerate}

\PSsubsection{مصورسازی}

\lr{PMF} به‌صورت \textbf{خطوط عمودی} (ضربه‌ها) در هر مقدار ممکن نمایش داده می‌شود، که ارتفاع آن برابر با احتمال است.

\tcbset{PScodemode/.style={unbreakable}}

\begin{PSplaincode}

  |     |

  |  |  |
0.2|  |  |  |
  |  |  |  |
0.1|  |  |  |  |
  |  |  |  |  |
  +--+--+--+--+--→ x
     0  1  2  3  4
\end{Verbatim}
\end{PSplaincode}

8 بیت ارسال می‌شود که هر بیت احتمال خطای مستقل \lr{p = 0.1} دارد.\PSbr{}\lr{X} = تعداد خطاهای بیت.

\lr{p\textsubscript{X}(k) = C(8,k) \ensuremath{\cdot} (0.1)\textsuperscript{k} \ensuremath{\cdot} (0.9)\textsuperscript{8-k}, for k = 0, 1, …, 8}

\PSsection{2.6}{تابع چگالی احتمال \lr{(PDF)}}

\begin{PSbox}[unbreakable]{def}{تعریف}

برای یک متغیر تصادفی پیوسته \lr{X}، \textbf{تابع چگالی احتمال} \lr{f\textsubscript{X}(x)} به‌گونه‌ای تعریف می‌شود که:

\PSdisplay{P(a \leq X \leq b) = \int_a^b f_X(x) \, dx}

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{نامنفی بودن:} \lr{f\textsubscript{X}(x) \ensuremath{\ge} 0} برای تمام \lr{x}
\item
  \textbf{نرمال‌سازی:} \lr{\ensuremath{\int}\textsubscript{-\ensuremath{\infty}}\textsuperscript{\ensuremath{\infty}} f\textsubscript{X}(x) dx = 1}
\item
  \textbf{احتمال = مساحت زیر منحنی} میان دو نقطه
\end{enumerate}

\PSsubsection{نکته حیاتی: \lr{f\textsubscript{X}(x)} یک احتمال نیست!}

\begin{PSquote}

\lr{\PSwarn{} f\textsubscript{X}(x)} می‌تواند بزرگ‌تر از 1 باشد!

\end{PSquote}

\lr{PDF} \textbf{چگالی} احتمال را می‌دهد \lr{—} احتمال در واحد طول. تنها \textbf{انتگرال} (مساحت) احتمال را می‌دهد.

\textbf{قیاس:} چگالی جرم \lr{(kg/m\textsuperscript{3})} جرم نیست. برای به‌دست‌آوردن جرم باید چگالی را روی حجم انتگرال بگیرید. به‌همین‌ترتیب، چگالی احتمال هم احتمال نیست \lr{—} برای به‌دست‌آوردن احتمال باید روی یک بازه انتگرال بگیرید.

\PSsubsection{مصورسازی}

\tcbset{PScodemode/.style={unbreakable}}

\begin{PSplaincode}
\begin{Verbatim}
f_X(x)
  |
  |      ╭────╮
  |     ╱ ░░░░ ╲        ← Shaded area = P(a ≤ X ≤ b)
  |    ╱ ░░░░░░ ╲
  |   ╱ ░░░░░░░░ ╲
  |──╱───░░░░░░░░──╲──→ x
  |      a      b
  Total area under curve = 1
\end{Verbatim}
\end{PSplaincode}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: ولتاژ نویز حرارتی}

ولتاژ نویز حرارتی در یک مدار اغلب از توزیع گاوسی پیروی می‌کند:

\PSdisplay{f_X(x) = \frac{1}{\sigma\sqrt{2\pi}} e^{-x^2/(2\sigma^2)}}

که در آن \lr{\ensuremath{\sigma}} ولتاژ مؤثر \lr{(RMS)} نویز است.

\lr{P(noise exceeds 2\ensuremath{\sigma}) = P(\textbar{}X\textbar{} \textgreater{} 2\ensuremath{\sigma}) = 1 - P(-2\ensuremath{\sigma} \ensuremath{\le} X \ensuremath{\le} 2\ensuremath{\sigma}) \ensuremath{\approx} 1 - 0.9545 = 0.0455}

\end{PSbox}

\PSsection{2.7}{تابع توزیع تجمعی \lr{(CDF)}}

\begin{PSbox}[unbreakable]{def}{تعریف}

برای هر متغیر تصادفی \lr{X} (گسسته یا پیوسته)، \textbf{تابع توزیع تجمعی} عبارت است از:

\PSdisplay{F_X(x) = P(X \leq x)}

\end{PSbox}

\PSsubsection{ویژگی‌ها}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{نامنزولی:} اگر \lr{a \textless{} b}، آنگاه \lr{F\textsubscript{X}(a) \ensuremath{\le} F\textsubscript{X}(b)}
\item
  \textbf{حدود:} \lr{F\textsubscript{X}(-\ensuremath{\infty}) = 0}، \lr{F\textsubscript{X}(+\ensuremath{\infty}) = 1}
\item
  \textbf{پیوستگی از راست:} \lr{F\textsubscript{X}(x)} از سمت راست پیوسته است
\item
  \textbf{احتمال یک بازه:} \lr{P(a \textless{} X \ensuremath{\le} b) = F\textsubscript{X}(b) - F\textsubscript{X}(a)}
\end{enumerate}

\PSsubsection{\lr{CDF} برای متغیرهای تصادفی گسسته}

\lr{CDF} یک \textbf{تابع پله‌ای} با جهش‌هایی در هر مقدار ممکن است:

\tcbset{PScodemode/.style={unbreakable}}

\begin{PSplaincode}
\begin{Verbatim}
F_X(x)
  |
1.0|                          ──────────
  |                    ○─────┘
0.8|              ○────┘
  |         ○───┘
0.4|    ○───┘
  |────┘
0.0|
  +----+----+----+----+---→ x
       0    1    2    3
\end{Verbatim}
\end{PSplaincode}

جهش در \lr{x = k} دارای ارتفاع \lr{p\textsubscript{X}(k)} است.

\PSsubsection{\lr{CDF} برای متغیرهای تصادفی پیوسته}

\lr{CDF} یک منحنی \textbf{هموار و پیوسته} است:

\PSdisplay{F_X(x) = \int_{-\infty}^{x} f_X(t) \, dt}

و برعکس:\PSdisplay{f_X(x) = \frac{dF_X(x)}{dx}}

\tcbset{PScodemode/.style={unbreakable}}

\begin{PSplaincode}
\begin{Verbatim}
F_X(x)
  |
1.0|                         ___________
  |                    ╱
  |                 ╱
0.5|             ╱
  |          ╱
  |       ╱
0.0|______╱
  +---+---+---+---+---+---→ x
\end{Verbatim}
\end{PSplaincode}

\PSsubsection{قیاس بصری: انبار}

احتمال را به‌عنوان «ماده‌ای» تصور کنید که روی مقادیر ممکن توزیع شده است:

\textbf{گسسته \lr{(PMF)}:} مانند انبار کردن ماده در جعبه‌های شماره‌گذاری‌شده. هر جعبه یک جرم مشخص دارد. \lr{PMF} جرم هر جعبه را به شما می‌گوید.

\tcbset{PScodemode/.style={unbreakable}}

\begin{PSplaincode}
\begin{Verbatim}
    ┌───┐
    │0.4│ ┌───┐
    │   │ │0.3│ ┌───┐
    │   │ │   │ │0.2│ ┌───┐
    └───┘ └───┘ └───┘ └───┘
     x=0   x=1   x=2   x=3
     
    Total mass in all boxes = 1
\end{Verbatim}
\end{PSplaincode}

\textbf{پیوسته \lr{(PDF)}:} مانند پهن‌کردن ماده در طول یک قفسه پیوسته. چگالی به شما می‌گوید ماده در هر نقطه چقدر ضخیم است. جرم را با محاسبه مساحت (ضخامت \ensuremath{\times} عرض) به دست می‌آورید.

\tcbset{PScodemode/.style={unbreakable}}

\begin{PSplaincode}
\begin{Verbatim}
         ╭────────╮
        ╱          ╲
       ╱     ░░░░    ╲      ← Area of shaded region = P(1 ≤ X ≤ 2)
      ╱     ░░░░░░    ╲
    ─╱──────░░░░░░░░────╲──
          1      2

    Total area under curve = 1
\end{Verbatim}
\end{PSplaincode}

\textbf{تجمعی \lr{(CDF)}:} مانند یک جمع در حال اجرا \lr{—} تا این نقطه چه مقدار ماده انباشته‌اید؟

\PSsubsection{چرا این موضوع در مهندسی اهمیت دارد}

\textbf{مثال:} یک ولتاژ نویز \lr{X} دارای \lr{PDF f\textsubscript{X}(x) = 2} برای \lr{x \ensuremath{\in} [0, 0.5]} است.

\begin{itemize}
\tightlist
\item
  \lr{f\textsubscript{X}(0.3) = 2} \lr{—} این یک احتمال نیست! به این معناست که چگالی برابر \lr{2 V\textsuperscript{-1}} است.
\item
  \lr{P(0.2 \ensuremath{\le} X \ensuremath{\le} 0.4) = \ensuremath{\int}\textsubscript{0.2}\textsuperscript{0.4} 2 dx = 2 \ensuremath{\times} 0.2 = 0.4} \lr{—} این یک احتمال است.
\item
  \lr{P(X = 0.3) = 0} \lr{—} برای متغیرهای تصادفی پیوسته، یک نقطه منفرد احتمال صفر دارد.
\end{itemize}

\PSsection{2.9}{تبدیل‌های متغیرهای تصادفی}

\PSsubsection{انگیزه}

در مهندسی، اغلب توزیع یک کمیت را می‌دانیم اما توزیع \textbf{تابعی} از آن کمیت را نیاز داریم:

\begin{itemize}
\tightlist
\item
  ولتاژ \lr{V} معلوم است \lr{\ensuremath{\to}} توان \lr{P = V\textsuperscript{2}/R} لازم است
\item
  سیگنال خطی \lr{X} معلوم است \lr{\ensuremath{\to}} مقدار \lr{dB} یعنی \lr{Y = 10\ensuremath{\cdot}log\textsubscript{10}(X)} لازم است
\item
  نویز \lr{N} معلوم است \lr{\ensuremath{\to} \textbar{}N}\lr{\textbar{}} (نویز یکسوسازی‌شده) لازم است
\end{itemize}

\PSsubsection{بیان مسیله}

داده شده: \lr{X} با \lr{PDF} معلوم \lr{f\textsubscript{X}(x)} یا \lr{PMF} معلوم \lr{p\textsubscript{X}(x)}\PSbr{}مطلوب: توزیع \lr{Y = g(X)}

\PSsubsection{حالت 1: تبدیل متغیر تصادفی گسسته}

اگر \lr{X} گسسته با \lr{PMF p\textsubscript{X}(x)} باشد و \lr{Y = g(X)}:

\PSdisplay{p_Y(y) = \sum_{x: g(x)=y} p_X(x)}

احتمال‌های تمام مقادیر \lr{x} را که به همان \lr{y} نگاشت می‌شوند جمع کنید.

\textbf{مثال:} \lr{X \ensuremath{\in} \{-2, -1, 0, 1, 2\}} با احتمال برابر \PSnum{0.2} برای هر یک.\PSbr{}\lr{Y = X\textsuperscript{2}}

\PSsubsection{حالت 2: تبدیل یکنوا (پیوسته)}

اگر \lr{Y = g(X)} که در آن \lr{g} اکیداً یکنوا و مشتق‌پذیر است:

\PSdisplay{f_Y(y) = f_X(g^{-1}(y)) \cdot \left|\frac{dg^{-1}(y)}{dy}\right|}

یا به‌طور معادل:\PSdisplay{f_Y(y) = \frac{f_X(x)}{|g'(x)|}\bigg|_{x=g^{-1}(y)}}

\textbf{استخراج از طریق روش \lr{CDF}:}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  با \lr{CDF} شروع کنید: \lr{F\textsubscript{Y}(y) = P(Y \ensuremath{\le} y) = P(g(X) \ensuremath{\le} y)}
\item
  وارون کنید: = \lr{P(X \ensuremath{\le} g\textsuperscript{-1}(y))} اگر \lr{g} صعودی باشد
\item
  مشتق بگیرید: \lr{f\textsubscript{Y}(y) = f\textsubscript{X}(g\textsuperscript{-1}(y)) \ensuremath{\cdot} d/dy[g\textsuperscript{-1}(y)]}
\end{enumerate}

\PSsubsection{حالت 3: تبدیل غیریکنوا}

اگر \lr{g} یکنوا نباشد، دامنه را به بازه‌هایی تقسیم کنید که در آن‌ها \lr{g} یکنوا است:

\PSdisplay{f_Y(y) = \sum_i \frac{f_X(x_i)}{|g'(x_i)|}}

که در آن \lr{x\textsubscript{1}, x\textsubscript{2}}, \lr{…} همگی جواب‌های \lr{g(x) = y} هستند.

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: توان از ولتاژ}

یک ولتاژ نویز \lr{X} دارای \lr{PDF f\textsubscript{X}(x)} است (مثلاً گاوسی با میانگین صفر).\PSbr{}توان تلف‌شده در یک مقاومت \lr{1\ensuremath{\Omega}: P = X\textsuperscript{2}}

از آنجا که \lr{Y = X\textsuperscript{2}} یکنوا نیست (هر دو \lr{+x} و \lr{-x} همان \lr{y} را می‌دهند):

برای \lr{y \textgreater{} 0}: هر دو \lr{x = +\ensuremath{\surd}y} و \lr{x = -\ensuremath{\surd}y} جواب هستند.\PSbr{}\lr{g(x) = x\textsuperscript{2}}، پس \lr{g’(x) = 2x}

\PSdisplay{f_Y(y) = \frac{f_X(\sqrt{y})}{2\sqrt{y}} + \frac{f_X(-\sqrt{y})}{2\sqrt{y}}, \quad y > 0}

اگر \lr{X} گاوسی با میانگین صفر باشد: \lr{f\textsubscript{X}(x) = f\textsubscript{X}(-x)}، پس:\PSdisplay{f_Y(y) = \frac{f_X(\sqrt{y})}{\sqrt{y}}, \quad y > 0}

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال مهندسی: تقویت خطی}

سیگنال \lr{Y = aX + b} (تقویت‌کننده با بهره \lr{a} و آفست \lr{b}).\PSbr{}\lr{X} دارای \lr{PDF f\textsubscript{X}(x)} است.

\lr{g\textsuperscript{-1}(y) = (y-b)/a}، و \lr{\textbar{}}\lr{d/dy[(y-b)/a]\textbar{} = 1/\textbar{}a}\lr{\textbar{}}

\PSdisplay{f_Y(y) = \frac{1}{|a|} f_X\left(\frac{y-b}{a}\right)}

\end{PSbox}

\item
  گسسته، \lr{X \ensuremath{\in} \{0, 1, 2, …\}}
\item

\item

\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2: ویژگی‌های \lr{PMF}}

یک متغیر تصادفی \lr{X} دارای \lr{PMF} زیر است: \lr{p\textsubscript{X}(0) = 0.1}، \lr{p\textsubscript{X}(1) = 0.3}، \lr{p\textsubscript{X}(2) = c}، \lr{p\textsubscript{X}(3) = 0.2}.

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  مقدار \lr{c} را بیابید.
\item
  \lr{P(X \ensuremath{\ge} 2)} را بیابید.
\item
  \lr{F\textsubscript{X}(1.5)} را بیابید.
\item
  \lr{P(0.5 \textless{} X \ensuremath{\le} 2.5)} را بیابید.
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\item
  مجموع = \lr{1: 0.1 + 0.3 + c + 0.2 = 1 \ensuremath{\to} c = 0.4}
\item
  \lr{P(X \ensuremath{\ge} 2) = p\textsubscript{X}(2) + p\textsubscript{X}(3) = 0.4 + 0.2 = 0.6}
\item
  \lr{F\textsubscript{X}(1.5) = P(X \ensuremath{\le} 1.5) = p\textsubscript{X}(0) + p\textsubscript{X}(1) = 0.1 + 0.3 = 0.4}
\item
  \lr{P(0.5 \textless{} X \ensuremath{\le} 2.5) = p\textsubscript{X}(1) + p\textsubscript{X}(2) = 0.3 + 0.4 = 0.7}
\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله \lr{3: PDF} و \lr{CDF}}

یک متغیر تصادفی دارای \lr{PDF} زیر است:\PSbr{}\lr{f\textsubscript{X}(x) = cx} برای \lr{0 \ensuremath{\le} x \ensuremath{\le} 4}، و در غیر این صورت 0.

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  مقدار \lr{c} را بیابید.
\item
  تابع \lr{CDF} یعنی \lr{F\textsubscript{X}(x)} را بیابید.
\item
  \lr{P(1 \ensuremath{\le} X \ensuremath{\le} 3)} را بیابید.
\item
  مقدار \lr{x\textsubscript{0}} را بیابید که در آن \lr{P(X \ensuremath{\le} x\textsubscript{0}) = 0.5}.
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\item
  \lr{\ensuremath{\int}\textsubscript{0}\textsuperscript{4} cx dx = c\ensuremath{\cdot}[x\textsuperscript{2}/2]\textsubscript{0}\textsuperscript{4} = c\ensuremath{\cdot}8 = 1 \ensuremath{\to} c = 1/8}
\item
  \lr{F\textsubscript{X}(x) = \ensuremath{\int}\textsubscript{0}\textsuperscript{x} (t/8)dt = x\textsuperscript{2}/16}، برای \lr{0 \ensuremath{\le} x \ensuremath{\le} 4}\PSbr{}\lr{F\textsubscript{X}(x) = 0} برای \lr{x \textless{} 0}؛ \lr{F\textsubscript{X}(x) = 1} برای \lr{x \textgreater{} 4}
\item
  \lr{P(1 \ensuremath{\le} X \ensuremath{\le} 3) = F\textsubscript{X}(3) - F\textsubscript{X}(1) = 9/16 - 1/16 = 8/16 = 0.5}
\item
  \lr{x\textsubscript{0}\textsuperscript{2}/16 = 0.5 \ensuremath{\to} x\textsubscript{0}\textsuperscript{2} = 8 \ensuremath{\to} x\textsubscript{0} = 2\ensuremath{\surd}2 \ensuremath{\approx} 2.83}
\end{enumerate}

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 4: تبدیل}

یک حسگر دما \lr{X} را اندازه‌گیری می‌کند که به‌طور یکنواخت روی \lr{[20, 30] \ensuremath{^{\circ}}C} توزیع شده است.\PSbr{}ولتاژ خروجی \lr{Y = 0.1X - 2} است (حسگر خطی).

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  \lr{PDF} مربوط به \lr{Y} را بیابید.
\item
  محدوده \lr{Y} را بیابید.
\item
  \lr{P(Y \textgreater{} 0.5)} را بیابید.
\end{enumerate}

\PSsolution{} 

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\item
  \lr{X \textasciitilde{} Uniform[20, 30]}، پس \lr{f\textsubscript{X}(x) = 1/10} برای \lr{20 \ensuremath{\le} x \ensuremath{\le} 30}.\PSbr{}\lr{Y = 0.1X - 2}، پس \lr{a = 0.1} و \lr{b = -2}.\PSbr{}\lr{f\textsubscript{Y}(y) = (1/\textbar{}a\textbar{}) \ensuremath{\cdot} f\textsubscript{X}((y-b)/a) = (1/0.1) \ensuremath{\cdot} (1/10) = 1} برای محدوده معتبر.
\item
  وقتی \lr{X = 20: Y = 0.1(20) - 2 = 0}؛ وقتی \lr{X = 30: Y = 0.1(30) - 2 = 1}\PSbr{}محدوده \lr{Y: [0, 1]}. پس \lr{Y \textasciitilde{} Uniform[0, 1]}.
\item
  \lr{P(Y \textgreater{} 0.5) = 1 - F\textsubscript{Y}(0.5) = 1 - 0.5 = 0.5}
\end{enumerate}

\end{PSbox}
