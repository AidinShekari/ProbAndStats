\begin{PSbox}[unbreakable]{obj}{اهداف یادگیری}

پس از تکمیل این ماژول، دانشجویان قادر خواهند بود:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  فرض صفر و فرض مقابل را برای مسایل مهندسی صورت‌بندی کنند
\item
  خطاهای نوع اول و نوع دوم و پیامدهای مهندسی آن‌ها را تشخیص دهند
\item
  آماره آزمون و مقادیر \lr{p} را محاسبه کنند
\item
  آزمون‌های \lr{Z} و \lr{t} را برای میانگین‌ها انجام دهند
\item
  مقادیر \lr{p} را به‌درستی تفسیر کنند (و از تصورهای نادرست رایج پرهیز کنند)
\item
  میان آزمون‌های یک‌طرفه و دوطرفه انتخاب کنند
\end{enumerate}

\end{PSbox}

\PSsection{15.1}{مقدمه: تصمیم‌گیری از داده}

\textbf{پرسش مهندسی:} «آیا این حسگر اریب است؟» «آیا فرایند تغییر کرده است؟» «آیا الگوریتم جدید بهتر عمل می‌کند؟»

به یک \textbf{چارچوب نظام‌مند} نیاز داریم تا به این پرسش‌ها از روی داده پاسخ دهیم، در حالی که احتمال خطا را کنترل می‌کنیم.

\PSsection{15.2}{چارچوب}

\PSsubsection{فرض صفر \lr{H\textsubscript{0}}}

\textbf{فرض پیش‌فرض} \lr{—} معمولاً «هیچ چیز تغییر نکرده است» یا «هیچ اثری وجود ندارد»:

\begin{itemize}
\tightlist
\item
  \lr{H\textsubscript{0}: \ensuremath{\mu} = \ensuremath{\mu}\textsubscript{0}} (حسگر اریب نیست)
\item
  \lr{H\textsubscript{0}: \ensuremath{\mu}\textsubscript{1} = \ensuremath{\mu}\textsubscript{2}} (دو روش به یک اندازه عمل می‌کنند)
\item
  \lr{H\textsubscript{0}: \ensuremath{\sigma}\textsuperscript{2} = \ensuremath{\sigma}\textsubscript{0}\textsuperscript{2}} (واریانس تغییر نکرده است)
\end{itemize}

\PSsubsection{فرض مقابل \lr{H\textsubscript{1}}}

آنچه می‌خواهیم \textbf{آشکار کنیم} یا اثبات کنیم:

\begin{itemize}
\tightlist
\item
  \lr{H\textsubscript{1}: \ensuremath{\mu} \ensuremath{\neq} \ensuremath{\mu}\textsubscript{0}} (حسگر اریب است \lr{—} دوطرفه)
\item
  \lr{H\textsubscript{1}: \ensuremath{\mu} \textgreater{} \ensuremath{\mu}\textsubscript{0}} (عملکرد بهبود یافته است \lr{—} یک‌طرفه)
\item
  \lr{H\textsubscript{1}: \ensuremath{\sigma}\textsuperscript{2} \textgreater{} \ensuremath{\sigma}\textsubscript{0}\textsuperscript{2}} (واریانس افزایش یافته است)
\end{itemize}

\PSsubsection{تصمیم}

بر اساس داده، یکی از این دو:

\begin{itemize}
\tightlist
\item
  \textbf{رد \lr{H\textsubscript{0}}} \lr{\ensuremath{\to}} شواهد از \lr{H\textsubscript{1}} پشتیبانی می‌کنند
\item
  \textbf{عدم رد \lr{H\textsubscript{0}}} \lr{\ensuremath{\to}} شواهد ناکافی علیه \lr{H\textsubscript{0}}
\end{itemize}

\begin{PSquote}

\lr{\PSwarn{}} «عدم رد \lr{H\textsubscript{0}» \ensuremath{\neq} «H\textsubscript{0}} درست است». نبودِ شواهد \lr{\ensuremath{\neq}} شواهدِ نبود.

\end{PSquote}

\begin{longtable}[c]{@{}ccc@{}}

\endhead
\bottomrule\noalign{}
\endlastfoot

\end{longtable}

\PSsubsection{خطای نوع اول (هشدار کاذب) \lr{—} با احتمال \lr{\ensuremath{\alpha}}}

رد \lr{H\textsubscript{0}} هنگامی که در واقع درست است. «فریاد گرگ.»

قیاس مهندسی: اعلام تغییر یک فرایند در حالی که تغییر نکرده است \lr{\ensuremath{\to}} کالیبراسیون مجدد بی‌مورد.

\PSsubsection{خطای نوع دوم (آشکارسازی از‌دست‌رفته) \lr{—} با احتمال \lr{\ensuremath{\beta}}}

عدم رد \lr{H\textsubscript{0}} هنگامی که \lr{H\textsubscript{1}} در واقع درست است. «از دست دادن سیگنال.»

قیاس مهندسی: عدم آشکارسازی اریبی یک حسگر \lr{\ensuremath{\to}} ادامه اندازه‌گیری‌های اریب.

\PSsubsection{قیاس مهندسی: آشکارسازی رادار}

\begin{longtable}[c]{@{}cc@{}}
\toprule\noalign{}
\rowcolor{PSnavy}\PSth{اصطلاح آماری} & \PSth{معادل رادار} \\
\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
\lr{H\textsubscript{0}}: هیچ هدفی نیست & هیچ هواپیمایی حاضر نیست \\
\lr{H\textsubscript{1}}: هدف حاضر است & هواپیما حاضر است \\

\end{longtable}
}

\PSsubsection{توازن}

کاهش \lr{\ensuremath{\alpha}} (هشدارهای کاذب کمتر) \lr{\ensuremath{\to}} افزایش \lr{\ensuremath{\beta}} (آشکارسازی‌های از‌دست‌رفته بیشتر)، و برعکس.

\lr{\ensuremath{\alpha}} را ثابت کنید (سطح معناداری)، سپس توان = \lr{1 - \ensuremath{\beta}} را بیشینه کنید.

\PSsection{15.4}{سطح معناداری \lr{\ensuremath{\alpha}}}

\textbf{حداکثر نرخ قابل‌قبول خطای نوع اول}، که \textbf{پیش از} نگاه‌کردن به داده انتخاب می‌شود.

انتخاب‌های رایج: \lr{\ensuremath{\alpha} = 0.05}، \PSnum{0.01}، \PSnum{0.10}

\textbf{انتخاب \lr{\ensuremath{\alpha}}:} هزینه هشدار کاذب را در نظر بگیرید:

\begin{itemize}
\tightlist
\item
  \lr{\ensuremath{\alpha} = 0.01}: بسیار محافظه‌کارانه (مثلاً ایمنی تجهیزات پزشکی)
\item
  \lr{\ensuremath{\alpha} = 0.05}: استاندارد در بیشتر شاخه‌های مهندسی/علوم
\item
  \lr{\ensuremath{\alpha} = 0.10}: با رواداری بیشتر (تحلیل اکتشافی)
\end{itemize}

\PSsection{15.5}{توان یک آزمون}

\PSdisplay{\PSmtext{Power} = 1 - \beta = P(\PSmtext{reject } H_0 | H_1 \PSmtext{ is true})}

توان به موارد زیر بستگی دارد:

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{اندازه اثر:} تفاوت واقعی بزرگ‌تر \lr{\ensuremath{\to}} آشکارسازی آسان‌تر \lr{\ensuremath{\to}} توان بیشتر
\item
  \textbf{اندازه نمونه \lr{n}:} داده بیشتر \lr{\ensuremath{\to}} توان بیشتر
\item
  \textbf{سطح معناداری \lr{\ensuremath{\alpha}}:} \lr{\ensuremath{\alpha}} بزرگ‌تر \lr{\ensuremath{\to}} توان بیشتر (اما هشدار کاذب بیشتر)
\item
  \textbf{تغییرپذیری \lr{\ensuremath{\sigma}}:} نویز کمتر \lr{\ensuremath{\to}} توان بیشتر
\end{enumerate}

\PSsection{15.6}{مقدار \lr{p}}

\begin{PSbox}[unbreakable]{def}{تعریف}

\textbf{مقدار \lr{p}} احتمال به‌دست‌آوردن یک آماره آزمون \textbf{به‌اندازه مشاهده‌شده حدی یا حدی‌تر} است، \textbf{با فرض درست‌بودن \lr{H\textsubscript{0}}}.

\PSdisplay{p\PSmtext{-value} = P(\PSmtext{data this extreme or more} | H_0 \PSmtext{ true})}

\end{PSbox}

\PSsubsection{قاعده تصمیم}

\begin{itemize}
\tightlist
\item
  اگر مقدار \lr{p \textless{} \ensuremath{\alpha} \ensuremath{\to} H\textsubscript{0}} را رد کنید
\item
  اگر مقدار \lr{p \ensuremath{\ge} \ensuremath{\alpha} \ensuremath{\to} H\textsubscript{0}} را رد نکنید
\end{itemize}

\PSsubsection{\lr{\PSyes{}} تفسیر درست}

\begin{PSquote}

«مقدار \lr{p} احتمال مشاهده داده‌ای به این حدی است (یا حدی‌تر) اگر \lr{H\textsubscript{0}} درست بود. مقدار \lr{p} کوچک به این معناست که داده تحت \lr{H\textsubscript{0}} نامحتمل است.»

\end{PSquote}

\PSsubsection{\lr{\PSno{}} تفسیر نادرست}

\begin{PSquote}

{«مقدار \lr{p} احتمال درست‌بودن \lr{H\textsubscript{0}} است.»}

\end{PSquote}

\textbf{چرا نادرست است:} \lr{H\textsubscript{0}} یا درست است یا نادرست (ثابت). مقدار \lr{p} هیچ چیزی درباره \lr{P(H\textsubscript{0} true)} نمی‌گوید. این مقدار می‌سنجد که داده تحت \lr{H\textsubscript{0}} چقدر شگفت‌آور است.

\PSsubsection{درجه‌بندی}

\midrule\noalign{}
\endhead
\bottomrule\noalign{}
\endlastfoot
\lr{\textgreater{}} \PSnum{0.10} & ضعیف یا هیچ \\

\PSsection{15.7}{آزمون‌های یک‌طرفه در برابر دوطرفه}

\PSsubsection{دوطرفه \lr{(H\textsubscript{1}: \ensuremath{\mu} \ensuremath{\neq} \ensuremath{\mu}\textsubscript{0})}}

هنگامی به کار می‌رود که انحراف در \textbf{هر} جهت مهم باشد.

\begin{itemize}
\tightlist
\item
  رد کنید اگر \lr{\textbar{}}\lr{T\textbar{} \textgreater{} t\_\{\ensuremath{\alpha}/2, n-1\}}
\item
  مقدار \lr{p = 2\ensuremath{\cdot}P(T \textgreater{} \textbar{}t\textsubscript{obs}\textbar{})}
\end{itemize}

\PSsubsection{یک‌طرفه (\lr{H\textsubscript{1}: \ensuremath{\mu} \textgreater{} \ensuremath{\mu}\textsubscript{0}} یا \lr{H\textsubscript{1}: \ensuremath{\mu} \textless{} \ensuremath{\mu}\textsubscript{0}})}

هنگامی به کار می‌رود که تنها یک جهت انحراف اهمیت دارد.

\begin{itemize}
\tightlist
\item
  \lr{H\textsubscript{1}: \ensuremath{\mu} \textgreater{} \ensuremath{\mu}\textsubscript{0}}: رد کنید اگر \lr{T \textgreater{} t\_\{\ensuremath{\alpha}, n-1\}}
\item
  \lr{H\textsubscript{1}: \ensuremath{\mu} \textless{} \ensuremath{\mu}\textsubscript{0}}: رد کنید اگر \lr{T \textless{} -t\_\{\ensuremath{\alpha}, n-1\}}
\end{itemize}

\PSsection{15.8}{آزمون‌های میانگین}

\PSsubsection{آزمون \lr{Z} تک‌نمونه‌ای (\lr{\ensuremath{\sigma}} معلوم)}

آزمون: \lr{H\textsubscript{0}: \ensuremath{\mu} = \ensuremath{\mu}\textsubscript{0}}

آماره آزمون: \lr{Z = (\ensuremath{\bar{X}} - \ensuremath{\mu}\textsubscript{0})/(\ensuremath{\sigma}/\ensuremath{\surd}n)}

تحت \lr{H\textsubscript{0}: Z \textasciitilde{} N(0,1)}

\PSsubsection{آزمون \lr{t} تک‌نمونه‌ای (\lr{\ensuremath{\sigma}} نامعلوم)}

آزمون: \lr{H\textsubscript{0}: \ensuremath{\mu} = \ensuremath{\mu}\textsubscript{0}}

آماره آزمون: \lr{T = (\ensuremath{\bar{X}} - \ensuremath{\mu}\textsubscript{0})/(S/\ensuremath{\surd}n)}

تحت \lr{H\textsubscript{0}: T \textasciitilde{} t(n-1)}

\PSsubsection{آزمون \lr{t} دو‌نمونه‌ای (مقایسه دو میانگین)}

آزمون: \lr{H\textsubscript{0}: \ensuremath{\mu}\textsubscript{1} = \ensuremath{\mu}\textsubscript{2}}

آماره آزمون: \lr{T = (\ensuremath{\bar{X}}\textsubscript{1} - \ensuremath{\bar{X}}\textsubscript{2}) / \ensuremath{\surd}(S\textsubscript{1}\textsuperscript{2}/n\textsubscript{1} + S\textsubscript{2}\textsuperscript{2}/n\textsubscript{2})}

درجات آزادی تقریبی بر اساس فرمول ولچ-ساترث‌ویت.

\PSsection{15.9}{آزمون واریانس (آزمون کای‌دو)}

\PSsubsection{آزمون تک‌نمونه‌ای}

\lr{H\textsubscript{0}: \ensuremath{\sigma}\textsuperscript{2} = \ensuremath{\sigma}\textsubscript{0}\textsuperscript{2}}

آماره آزمون: \lr{\ensuremath{\chi}\textsuperscript{2} = (n-1)S\textsuperscript{2}/\ensuremath{\sigma}\textsubscript{0}\textsuperscript{2}}

تحت \lr{H\textsubscript{0}: \ensuremath{\chi}\textsuperscript{2} \textasciitilde{} \ensuremath{\chi}\textsuperscript{2}(n-1)}

\PSsubsection{تصمیم}

\begin{itemize}
\tightlist
\item
  \lr{H\textsubscript{1}: \ensuremath{\sigma}\textsuperscript{2} \textgreater{} \ensuremath{\sigma}\textsubscript{0}\textsuperscript{2}}: رد کنید اگر \lr{\ensuremath{\chi}\textsuperscript{2} \textgreater{} \ensuremath{\chi}\textsuperscript{2}\_\{\ensuremath{\alpha}, n-1\}}
\item
  \lr{H\textsubscript{1}: \ensuremath{\sigma}\textsuperscript{2} \ensuremath{\neq} \ensuremath{\sigma}\textsubscript{0}\textsuperscript{2}}: رد کنید اگر \lr{\ensuremath{\chi}\textsuperscript{2} \textless{} \ensuremath{\chi}\textsuperscript{2}\_\{1-\ensuremath{\alpha}/2, n-1\}} یا \lr{\ensuremath{\chi}\textsuperscript{2} \textgreater{} \ensuremath{\chi}\textsuperscript{2}\_\{\ensuremath{\alpha}/2, n-1\}}
\end{itemize}

\PSsection{15.10}{رویه گام‌به‌گام آزمون فرض}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{بیان فرض‌ها:} \lr{H\textsubscript{0}} و \lr{H\textsubscript{1}}
\item
  \textbf{انتخاب سطح معناداری:} \lr{\ensuremath{\alpha}} (مثلاً \PSnum{0.05)}
\item
  \textbf{انتخاب آماره آزمون:} \lr{Z}، \lr{t} یا \lr{\ensuremath{\chi}\textsuperscript{2}} بسته به سناریو
\item
  \textbf{تعیین مقدار(های) بحرانی} یا محاسبه مقدار \lr{p}
\item
  \textbf{محاسبه آماره آزمون} از داده
\item
  \textbf{اتخاذ تصمیم:} رد \lr{H\textsubscript{0}} اگر آماره آزمون در ناحیه بحرانی باشد (یا \lr{p \textless{} \ensuremath{\alpha}})
\item
  \textbf{بیان نتیجه‌گیری} به زبان مهندسی
\end{enumerate}

\PSsection{15.11}{مثال‌های مهندسی}

\begin{PSbox}[unbreakable]{ex}{مثال 1: آیا یک حسگر اریب است؟}

یک حسگر باید بدون ورودی مقدار \lr{0V} را بخواند. 20 اندازه‌گیری \lr{\ensuremath{\bar{X}} = 0.012V} و \lr{S = 0.03V} می‌دهد.

\lr{H\textsubscript{0}: \ensuremath{\mu} = 0} (بدون اریبی)، \lr{H\textsubscript{1}: \ensuremath{\mu} \ensuremath{\neq} 0} (اریب)، \lr{\ensuremath{\alpha} = 0.05}

\lr{T = (0.012 - 0)/(0.03/\ensuremath{\surd}20) = 0.012/0.00671 = 1.789}

\lr{t\_\{0.025, 19\} = 2.093}. از آنجا که \lr{\textbar{}}\lr{T\textbar{} = 1.789 \textless{} 2.093 \ensuremath{\to}} \textbf{\lr{H\textsubscript{0}} را رد نکنید.}

مقدار \lr{p = 2\ensuremath{\cdot}P(t(19) \textgreater{} 1.789) \ensuremath{\approx} 0.090}. در سطح 5\% معنادار نیست.

نتیجه‌گیری: شواهد کافی برای نتیجه‌گیری اریب‌بودن حسگر وجود ندارد.

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال 2: آیا فرایند تولید تغییر کرده است؟}

پیش از تغییر: \lr{\ensuremath{\mu}\textsubscript{0} = 100\ensuremath{\Omega}} (تثبیت‌شده). پس از اصلاح: \lr{n = 30}، \lr{\ensuremath{\bar{X}} = 101.2\ensuremath{\Omega}}، \lr{S = 3.5\ensuremath{\Omega}}.

\lr{H\textsubscript{0}: \ensuremath{\mu} = 100}، \lr{H\textsubscript{1}: \ensuremath{\mu} \ensuremath{\neq} 100}، \lr{\ensuremath{\alpha} = 0.05}

\lr{T = (101.2 - 100)/(3.5/\ensuremath{\surd}30) = 1.2/0.639 = 1.878}

\lr{t\_\{0.025, 29\} = 2.045. \textbar{}T\textbar{} = 1.878 \textless{} 2.045 \ensuremath{\to}} \textbf{\lr{H\textsubscript{0}} را رد نکنید.}

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال 3: آیا الگوریتم جدید \lr{BER} را بهبود می‌دهد؟}

الگوریتم قدیمی: \lr{BER\textsubscript{0} = 10\textsuperscript{-3}}. الگوریتم جدید آزمایش‌شده: \lr{n = 50000} بیت، 38 خطا.

\lr{BER\textsubscript{new} = 38/50000 = 7.6\ensuremath{\times}10\textsuperscript{-4}}

\lr{H\textsubscript{0}: p = 0.001}، \lr{H\textsubscript{1}: p \textless{} 0.001} (بهبود)، \lr{\ensuremath{\alpha} = 0.05}

\lr{Z = (\ensuremath{\hat{p}} - p\textsubscript{0})/\ensuremath{\surd}(p\textsubscript{0}(1-p\textsubscript{0})/n) = (0.00076 - 0.001)/\ensuremath{\surd}(0.001\ensuremath{\times}0.999/50000) = -0.00024/0.000141 = -1.70}

مقدار \lr{p = P(Z \textless{} -1.70) = 0.0446 \textless{} 0.05 \ensuremath{\to}} \textbf{\lr{H\textsubscript{0}} را رد کنید.}

نتیجه‌گیری: شواهد از این که الگوریتم جدید \lr{BER} کمتری دارد پشتیبانی می‌کند.

\end{PSbox}

\begin{PSbox}[unbreakable]{ex}{مثال 4: آیا توان نویز افزایش یافته است؟}

مقدار تاریخی: \lr{\ensuremath{\sigma}\textsubscript{0}\textsuperscript{2} = 0.04 V\textsuperscript{2}}. اندازه‌گیری‌های جدید: \lr{n = 25}، \lr{S\textsuperscript{2} = 0.058 V\textsuperscript{2}}.

\lr{H\textsubscript{0}: \ensuremath{\sigma}\textsuperscript{2} = 0.04}، \lr{H\textsubscript{1}: \ensuremath{\sigma}\textsuperscript{2} \textgreater{} 0.04}، \lr{\ensuremath{\alpha} = 0.05}

\lr{\ensuremath{\chi}\textsuperscript{2} = (24)(0.058)/0.04 = 34.8}

\lr{\ensuremath{\chi}\textsuperscript{2}\_\{0.05, 24\} = 36.415}. از آنجا که \lr{34.8 \textless{} 36.415 \ensuremath{\to}} \textbf{\lr{H\textsubscript{0}} را رد نکنید.}

\end{PSbox}

\PSsection{15.13}{مسایل تمرینی}

\begin{PSbox}[unbreakable]{prob}{مسیله 1}

یک مشخصه ساخت و تولید میانگین مقاومت = \lr{100\ensuremath{\Omega}} را می‌طلبد. یک نمونه 16 مقاومتی \lr{\ensuremath{\bar{X}} = 101.5\ensuremath{\Omega}} و \lr{S = 3\ensuremath{\Omega}} می‌دهد. در \lr{\ensuremath{\alpha} = 0.05}، آیا شاهدی بر رانش فرایند وجود دارد؟

\PSsolution{} \lr{H\textsubscript{0}: \ensuremath{\mu}=100}، \lr{H\textsubscript{1}: \ensuremath{\mu}\ensuremath{\neq}100. T = (101.5-100)/(3/4) = 2.0. t\textsubscript{0.025,15} = 2.131. \textbar{}T\textbar{}=2.0 \textless{} 2.131 \ensuremath{\to}} رد نکنید. مقدار \lr{p \ensuremath{\approx} 0.064 \textgreater{} 0.05}.

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 2}

میانگین تأخیر بسته \lr{5ms} بود. پس از ارتقای شبکه، 40 اندازه‌گیری \lr{\ensuremath{\bar{X}} = 4.6ms} و \lr{S = 1.2ms} می‌دهد. در \lr{\ensuremath{\alpha} = 0.01} آزمون کنید که آیا تأخیر کاهش یافته است.

\PSsolution{} \lr{H\textsubscript{0}: \ensuremath{\mu}=5}، \lr{H\textsubscript{1}: \ensuremath{\mu}\textless{}5. T = (4.6-5)/(1.2/\ensuremath{\surd}40) = -0.4/0.190 = -2.11. t\textsubscript{0.01,39} \ensuremath{\approx} -2.426. T = -2.11 \textgreater{} -2.426 \ensuremath{\to}} در سطح 1\% رد نکنید. (در سطح 5\% رد می‌شد چون \lr{t\textsubscript{0.05,39}\ensuremath{\approx}-1.685}.)

\end{PSbox}

\begin{PSbox}[unbreakable]{prob}{مسیله 3}

توضیح دهید چرا «\lr{p = 0.04} به این معناست که تنها 4\% شانس وجود دارد که \lr{H\textsubscript{0}} درست باشد» \textbf{نادرست} است.

\PSsolution{} مقدار \lr{p} همان \lr{P(\rl{داده‌ای به این حدی \lr{\textbar{}}} H\textsubscript{0} true)} است، نه \lr{P(H\textsubscript{0} true \textbar{} \rl{داده})}. این مقدار به \lr{H\textsubscript{0}} درست‌بودن مشروط می‌شود و درباره حدی‌بودن داده می‌پرسد. برای به‌دست‌آوردن \lr{P(H\textsubscript{0} true \textbar{} \rl{داده})} به تحلیل بیزی با یک پیشین نیاز است. مقدار \lr{p} ویژگی فرایند تولید داده تحت \lr{H\textsubscript{0}} است، نه احتمال پسینی درباره \lr{H\textsubscript{0}}.
